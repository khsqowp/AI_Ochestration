(async () => {
  /*
   * XSS Scout v3.3 add-on: passive DOM, controlled-form-marker, request-shape,
   * and navigation observer.
   * It records only metadata and masked previews. It does not inject a
   * payload, navigate, alter request headers, or modify application state.
   * Reload the page to remove the temporary observers.
   */
  const __XSS_SCOUT_DOM__ = (() => {
    const MAX_EVENTS = 250;
    const events = [];
    const requestShapes = [];
    const formMarkers = new Set();
    const xhrMeta = new WeakMap();
    const restores = [];
    const listeners = [];
    const hookStatus = Object.create(null);
    const startedAt = Date.now();

    const mask = value => {
      const text = String(value ?? '');
      if (!text) return '';
      return `[masked:${text.length}]`;
    };

    const sourceValues = () => {
      const values = new Set();
      try {
        const u = new URL(location.href);
        for (const value of u.searchParams.values()) if (value) values.add(value);
        if (u.hash.length > 1) values.add(decodeURIComponent(u.hash.slice(1)));
      } catch { /* malformed location values are ignored */ }
      for (const marker of formMarkers) values.add(marker);
      return [...values].filter(value => value.length >= 3 && value.length <= 4096);
    };

    const routeLabel = value => {
      try {
        const u = new URL(value, location.href);
        const names = [...new Set(u.searchParams.keys())];
        return `${u.origin}${u.pathname}${names.length ? `?${names.map(name => `${encodeURIComponent(name)}=[redacted]`).join('&')}` : ''}`;
      } catch { return '[unparseable-url]'; }
    };

    const bodyShape = body => {
      if (body === undefined || body === null || body === '') return 'NONE';
      if (body instanceof FormData) return `FORM_DATA(keys:${[...body.keys()].map(String).slice(0, 20).join(',')})`;
      if (body instanceof URLSearchParams) return `URL_ENCODED(keys:${[...body.keys()].map(String).slice(0, 20).join(',')})`;
      if (typeof body !== 'string') return `NON_TEXT(${body.constructor?.name || typeof body})`;
      try {
        const parsed = JSON.parse(body);
        if (parsed && typeof parsed === 'object') return `JSON(keys:${Object.keys(parsed).slice(0, 20).join(',')})`;
      } catch { /* not JSON */ }
      if (/^[^=&]+=[^&]*(?:&|$)/.test(body)) {
        try { return `URL_ENCODED(keys:${[...new URLSearchParams(body).keys()].slice(0, 20).join(',')})`; } catch { /* fall through */ }
      }
      return `TEXT(length:${body.length})`;
    };

    const addRequestShape = (transport, method, value, body) => {
      if (requestShapes.length >= MAX_EVENTS) return;
      let sameOrigin = false;
      try { sameOrigin = new URL(value, location.href).origin === location.origin; } catch { /* handled by route label */ }
      if (!sameOrigin) return;
      requestShapes.push({
        Time: new Date().toISOString(), Transport: transport,
        Method: String(method || 'GET').toUpperCase(), URL: routeLabel(value),
        BodyShape: bodyShape(body),
      });
    };

    const sourceMatch = value => {
      const text = String(value ?? '');
      const matched = sourceValues().some(source => text.includes(source));
      return {
        matched,
        scoutMarker: /xss(?:scout|test)/i.test(text),
      };
    };

    const add = (kind, detail = {}) => {
      if (events.length >= MAX_EVENTS) return;
      const probe = sourceMatch(detail.value);
      events.push({
        Time: new Date().toISOString(),
        Kind: kind,
        Target: detail.target || '',
        Attribute: detail.attribute || '',
        Value: mask(detail.value),
        SourceMatch: probe.matched,
        ScoutMarker: probe.scoutMarker,
        Origin: detail.origin || '',
        CrossOrigin: Boolean(detail.crossOrigin),
      });
    };

    const patchMethod = (object, name, handler, label = name) => {
      const original = object?.[name];
      if (typeof original !== 'function') {
        hookStatus[label] = 'UNAVAILABLE';
        return false;
      }
      try {
        const wrapped = function (...args) {
          try { handler.call(this, args); } catch { /* observation must never break page code */ }
          return original.apply(this, args);
        };
        object[name] = wrapped;
        // sloppy mode에서 non-writable 속성에 대입하면 에러 없이 조용히 무시될 수
        // 있다 -- 대입이 실제로 먹었는지 확인하지 않으면 hookStatus가 ACTIVE라고
        // 거짓 보고하는데 실제로는 원본 메서드가 그대로 호출되는 상태가 된다.
        if (object[name] !== wrapped) {
          hookStatus[label] = 'UNAVAILABLE';
          return false;
        }
        restores.push(() => { if (object[name] === wrapped) object[name] = original; });
        hookStatus[label] = 'ACTIVE';
        return true;
      } catch {
        hookStatus[label] = 'UNAVAILABLE';
        return false;
      }
    };

    const patchSetter = (prototype, name, kind, label = name) => {
      try {
        const descriptor = Object.getOwnPropertyDescriptor(prototype, name);
        if (!descriptor?.set || !descriptor.configurable) {
          hookStatus[label] = 'UNAVAILABLE';
          return false;
        }
        const wrappedSetter = function (value) {
          add(kind, { target: this?.tagName || prototype.constructor?.name || '', value });
          return descriptor.set.call(this, value);
        };
        Object.defineProperty(prototype, name, {
          configurable: true,
          enumerable: descriptor.enumerable,
          get: descriptor.get,
          set: wrappedSetter,
        });
        restores.push(() => {
          if (Object.getOwnPropertyDescriptor(prototype, name)?.set === wrappedSetter) {
            Object.defineProperty(prototype, name, descriptor);
          }
        });
        hookStatus[label] = 'ACTIVE';
        return true;
      } catch {
        hookStatus[label] = 'UNAVAILABLE';
        return false;
      }
    };

    patchSetter(Element.prototype, 'innerHTML', 'HTML_SINK_INNER_HTML', 'Element.innerHTML');
    patchSetter(Element.prototype, 'outerHTML', 'HTML_SINK_OUTER_HTML', 'Element.outerHTML');
    if (typeof ShadowRoot !== 'undefined') {
      patchSetter(ShadowRoot.prototype, 'innerHTML', 'HTML_SINK_SHADOW_ROOT_INNER_HTML', 'ShadowRoot.innerHTML');
    }
    patchMethod(Element.prototype, 'insertAdjacentHTML', function (args) {
      add('HTML_SINK_INSERT_ADJACENT_HTML', { target: this?.tagName || '', value: args[1] });
    }, 'Element.insertAdjacentHTML');
    patchMethod(Element.prototype, 'setAttribute', function (args) {
      const attribute = String(args[0] || '').toLowerCase();
      if (attribute.startsWith('on') || ['srcdoc', 'href', 'src', 'action', 'formaction'].includes(attribute)) {
        add('ATTRIBUTE_OR_URL_SINK', { target: this?.tagName || '', attribute, value: args[1] });
      }
    }, 'Element.setAttribute');
    patchMethod(Document.prototype, 'write', function (args) {
      add('HTML_SINK_DOCUMENT_WRITE', { target: 'DOCUMENT', value: args.join('') });
    }, 'Document.write');
    patchMethod(Document.prototype, 'writeln', function (args) {
      add('HTML_SINK_DOCUMENT_WRITELN', { target: 'DOCUMENT', value: args.join('') });
    }, 'Document.writeln');
    patchMethod(globalThis.DOMParser?.prototype, 'parseFromString', function (args) {
      add('HTML_SINK_DOMPARSER', { target: 'DOMPARSER', value: args[0] });
    }, 'DOMParser.parseFromString');
    patchMethod(globalThis.Range?.prototype, 'createContextualFragment', function (args) {
      add('HTML_SINK_CONTEXTUAL_FRAGMENT', { target: 'RANGE', value: args[0] });
    }, 'Range.createContextualFragment');
    patchMethod(Element.prototype, 'setHTMLUnsafe', function (args) {
      add('HTML_SINK_SET_HTML_UNSAFE', { target: this?.tagName || '', value: args[0] });
    }, 'Element.setHTMLUnsafe');
    if (typeof ShadowRoot !== 'undefined') {
      patchMethod(ShadowRoot.prototype, 'setHTMLUnsafe', function (args) {
        add('HTML_SINK_SET_HTML_UNSAFE', { target: 'SHADOW_ROOT', value: args[0] });
      }, 'ShadowRoot.setHTMLUnsafe');
    }
    patchMethod(window, 'open', function (args) {
      const value = args[0];
      let crossOrigin = false;
      try { crossOrigin = value ? new URL(value, location.href).origin !== location.origin : false; } catch { /* invalid URL */ }
      add('WINDOW_OPEN', { target: 'WINDOW', value, crossOrigin });
    }, 'window.open');
    patchMethod(window, 'setTimeout', function (args) {
      if (typeof args[0] === 'string') add('STRING_TIMER_SINK', { target: 'WINDOW', value: args[0] });
    }, 'window.setTimeout');
    patchMethod(window, 'setInterval', function (args) {
      if (typeof args[0] === 'string') add('STRING_TIMER_SINK', { target: 'WINDOW', value: args[0] });
    }, 'window.setInterval');

    const messageHandler = event => {
      const value = typeof event.data === 'string' ? event.data : JSON.stringify(Object.keys(event.data || {}));
      add('POSTMESSAGE_RECEIVED', {
        target: 'WINDOW', value, origin: event.origin,
        crossOrigin: Boolean(event.origin && event.origin !== location.origin),
      });
    };
    window.addEventListener('message', messageHandler, true);
    listeners.push(() => window.removeEventListener('message', messageHandler, true));

    const registerControlledMarker = event => {
      const target = event.target;
      if (!(target instanceof HTMLInputElement || target instanceof HTMLTextAreaElement || target instanceof HTMLSelectElement)) return;
      if (target instanceof HTMLInputElement && target.type === 'password') return;
      const value = String(target.value || '');
      const markers = value.match(/\bXSSSCOUT_[A-Z0-9_-]{3,128}\b/gi) || [];
      for (const marker of markers) formMarkers.add(marker);
      if (markers.length) add('FORM_CONTROLLED_MARKER', { target: target.tagName, attribute: target.name || target.id || '', value });
    };
    document.addEventListener('input', registerControlledMarker, true);
    listeners.push(() => document.removeEventListener('input', registerControlledMarker, true));
    document.addEventListener('change', registerControlledMarker, true);
    listeners.push(() => document.removeEventListener('change', registerControlledMarker, true));
    const submitHandler = event => {
      event.target?.querySelectorAll?.('input, textarea, select').forEach(control => registerControlledMarker({ target: control }));
    };
    document.addEventListener('submit', submitHandler, true);
    listeners.push(() => document.removeEventListener('submit', submitHandler, true));
    document.querySelectorAll('input, textarea, select').forEach(control => registerControlledMarker({ target: control }));

    patchMethod(window, 'fetch', function (args) {
      const [input, init = {}] = args;
      const method = init.method || (input instanceof Request ? input.method : 'GET');
      const url = input instanceof Request ? input.url : input;
      addRequestShape('FETCH', method, url, init.body);
    }, 'window.fetch');
    patchMethod(XMLHttpRequest.prototype, 'open', function (args) {
      xhrMeta.set(this, { method: args[0], url: args[1] });
    }, 'XMLHttpRequest.open');
    patchMethod(XMLHttpRequest.prototype, 'send', function (args) {
      const meta = xhrMeta.get(this) || {};
      addRequestShape('XHR', meta.method || 'GET', meta.url || location.href, args[0]);
    }, 'XMLHttpRequest.send');

    const classify = event => {
      const htmlSink = event.Kind.startsWith('HTML_SINK');
      if (htmlSink && event.SourceMatch && event.ScoutMarker) return 'HIGH';
      if (htmlSink && event.SourceMatch) return 'MEDIUM';
      if (htmlSink || event.Kind === 'STRING_TIMER_SINK') return 'LOW';
      if (event.Kind === 'WINDOW_OPEN' && event.CrossOrigin) return 'LOW';
      return 'INFO';
    };

    return {
      version: '3.3', startedAt, events, requestShapes, hookStatus,
      summary() {
        const rows = events.map(event => ({ ...event, Signal: classify(event) }));
        return {
          total: rows.length,
          high: rows.filter(row => row.Signal === 'HIGH'),
          medium: rows.filter(row => row.Signal === 'MEDIUM'),
          low: rows.filter(row => row.Signal === 'LOW'),
          info: rows.filter(row => row.Signal === 'INFO'),
          rows,
          requestShapes: [...requestShapes],
          stateChangingRequestShapes: requestShapes.filter(row => !['GET', 'HEAD', 'OPTIONS'].includes(row.Method)),
          controlledInputMarkerCount: formMarkers.size,
          hookStatus: { ...hookStatus },
          activeHookCount: Object.values(hookStatus).filter(status => status === 'ACTIVE').length,
          unavailableHooks: Object.entries(hookStatus).filter(([, status]) => status !== 'ACTIVE').map(([hook]) => hook),
        };
      },
      stop() {
        listeners.splice(0).reverse().forEach(remove => remove());
        restores.splice(0).reverse().forEach(restore => restore());
      },
    };
  })();

  // ============================================================
  // PARAMETER
  // ============================================================

  function getParameters(url) {

    const params = [];

    for (
      const [name]
      of url.searchParams.entries()
    ) {

      if (!name) {
        continue;
      }

      params.push(name);
    }

    return [
      ...new Set(params)
    ];
  }


  function getOriginalValues(
    url,
    parameter
  ) {

    return url.searchParams.getAll(
      parameter
    );
  }


  // ============================================================
  // HTTP STATUS CLASS
  // ============================================================

  function classifyStatus(status) {

    if (
      status >= 200 &&
      status < 300
    ) {
      return 'SUCCESS';
    }

    if (
      status >= 300 &&
      status < 400
    ) {
      return 'REDIRECT';
    }

    if (
      status >= 400 &&
      status < 500
    ) {
      return 'CLIENT_ERROR';
    }

    if (
      status >= 500 &&
      status < 600
    ) {
      return 'SERVER_ERROR';
    }

    return 'OTHER';
  }


  function isErrorClass(statusClass) {

    return (
      statusClass === 'CLIENT_ERROR' ||
      statusClass === 'SERVER_ERROR'
    );
  }


  // ============================================================
  // CONTENT TYPE
  // ============================================================

  function normalizeContentType(contentType) {

    return String(
      contentType || ''
    )
      .split(';')[0]
      .trim()
      .toLowerCase();
  }


  function getContentTypeTransition(
    baselineType,
    testedType
  ) {

    const from =
      normalizeContentType(
        baselineType
      );

    const to =
      normalizeContentType(
        testedType
      );

    if (
      from === to
    ) {
      return 'STABLE';
    }

    return `${from || '(none)'} → ${to || '(none)'}`;
  }


  // ============================================================
  // FETCH
  // ============================================================

  async function fetchGET(url) {

    const controller =
      new AbortController();

    const timer =
      setTimeout(
        () => controller.abort(),
        CONFIG.TIMEOUT
      );

    try {

      return await fetch(
        url,
        {
          method:
            'GET',

          credentials:
            'include',

          cache:
            'no-store',

          redirect:
            'follow',

          signal:
            controller.signal
        }
      );

    } finally {

      clearTimeout(timer);
    }
  }


  // ============================================================
  // RESPONSE
  // ============================================================

  async function readResponse(response) {

    const contentType =
      response.headers.get(
        'content-type'
      ) || '';

    const body =
      await response.text();

    return {

      status:
        response.status,

      statusClass:
        classifyStatus(
          response.status
        ),

      ok:
        response.ok,

      redirected:
        response.redirected,

      finalURL:
        response.url,

      contentType,

      normalizedContentType:
        normalizeContentType(
          contentType
        ),

      body,

      length:
        body.length
    };
  }


  // ============================================================
  // CONTEXT
  // ============================================================

  function getContext(
    body,
    marker,
    before = 150,
    after = 300
  ) {

    const index =
      body.indexOf(marker);

    if (
      index === -1
    ) {
      return '(NOT FOUND)';
    }

    return body
      .substring(

        Math.max(
          0,
          index - before
        ),

        Math.min(
          body.length,
          index +
          marker.length +
          after
        )
      )

      .replace(/\r/g, '\\r')
      .replace(/\n/g, '\\n')
      .replace(/\t/g, '\\t');
  }


  // ============================================================
  // REFLECTION POSITIONS
  // ============================================================

  function findMarkerPositions(
    body,
    marker
  ) {

    const positions = [];

    let offset = 0;

    while (true) {

      const index =
        body.indexOf(
          marker,
          offset
        );

      if (
        index === -1
      ) {
        break;
      }

      positions.push(index);

      offset =
        index +
        marker.length;
    }

    return positions;
  }

  // ============================================================
  // REPRESENTATION LIST
  // ============================================================

  function getRepresentations(character) {

    const representations = [

      {
        status:
          'RAW',

        representation:
          character
      }
    ];

    const definitions =
      CHAR_ENCODINGS[character] || {};

    for (
      const [
        encoding,
        values
      ]
      of Object.entries(definitions)
    ) {

      for (
        const value
        of values
      ) {

        representations.push({

          status:
            encoding,

          representation:
            value
        });
      }
    }

    return representations;
  }


  // ============================================================
  // CHARACTER MATCH
  // ============================================================

  function matchCharacter(
    text,
    offset,
    character
  ) {

    const representations =
      getRepresentations(
        character
      );

    for (
      const item
      of representations
    ) {

      if (
        text.startsWith(
          item.representation,
          offset
        )
      ) {

        return {

          status:
            item.status,

          representation:
            item.representation,

          position:
            offset,

          skipped:
            0,

          consumed:
            item.representation.length
        };
      }
    }

    return null;
  }


  // ============================================================
  // RESYNC SEARCH
  // ============================================================

  function findCharacterAhead(
    text,
    offset,
    character
  ) {

    const representations =
      getRepresentations(
        character
      );

    const searchEnd =
      Math.min(
        text.length,
        offset +
        CONFIG.RESYNC_LOOKAHEAD
      );

    let best = null;

    for (
      const item
      of representations
    ) {

      const index =
        text.indexOf(
          item.representation,
          offset
        );

      if (
        index === -1 ||
        index >= searchEnd
      ) {

        continue;
      }

      if (
        !best ||
        index < best.position ||
        (
          index === best.position &&
          item.representation.length >
          best.representation.length
        )
      ) {

        best = {

          status:
            item.status,

          representation:
            item.representation,

          position:
            index,

          skipped:
            index - offset,

          consumed:
            item.representation.length
        };
      }
    }

    return best;
  }


  // ============================================================
  // NEXT CHARACTER RESYNC
  // ============================================================

  function findNextExpectedCharacter(
    text,
    offset,
    specialCharacters,
    currentIndex
  ) {

    let best = null;

    for (
      let i =
        currentIndex + 1;

      i <
        specialCharacters.length;

      i++
    ) {

      const character =
        specialCharacters[i];

      const match =
        findCharacterAhead(
          text,
          offset,
          character
        );

      if (!match) {
        continue;
      }

      if (
        !best ||
        match.position <
        best.match.position
      ) {

        best = {

          characterIndex:
            i,

          character,

          match
        };
      }
    }

    return best;
  }


  // ============================================================
  // RESYNC CHARACTER ANALYSIS
  // ============================================================

  function analyzeReflectionAt(
    body,
    marker,
    markerPosition
  ) {

    const start =
      markerPosition +
      marker.length;

    const segment =
      body.substring(
        start,
        Math.min(
          body.length,
          start +
          CONFIG.TRANSFORM_LOOKAHEAD
        )
      );

    const characters =
      [...CONFIG.SPECIAL];

    const map = [];

    let offset = 0;

    let resyncCount = 0;


    for (
      let i = 0;
      i < characters.length;
      i++
    ) {

      const character =
        characters[i];


      // ----------------------------------------------------------
      // 1. 현재 위치에서 정상 match
      // ----------------------------------------------------------

      const direct =
        matchCharacter(
          segment,
          offset,
          character
        );


      if (
        direct
      ) {

        map.push({

          Character:
            character,

          Status:
            direct.status,

          Representation:
            direct.representation,

          Position:
            direct.position,

          Skipped:
            0,

          Resynced:
            false
        });


        offset =
          direct.position +
          direct.consumed;


        continue;
      }


      // ----------------------------------------------------------
      // 2. 현재 문자가 약간 뒤에서 발견되는지 확인
      //
      // 단순히 뒤에서 발견됐다고 바로 해당 문자라고 판단하면
      // 주변 HTML/JSON 문자와 혼동할 수 있으므로,
      // RESYNC_LOOKAHEAD 안에서만 제한적으로 탐색한다.
      // ----------------------------------------------------------

      const ahead =
        findCharacterAhead(
          segment,
          offset,
          character
        );


      // ----------------------------------------------------------
      // 3. 다음 기대 문자 위치도 확인
      // ----------------------------------------------------------

      const nextExpected =
        findNextExpectedCharacter(
          segment,
          offset,
          characters,
          i
        );


      /*
       * 현재 문자의 representation이 먼저 나온다면:
       *
       *   unexpected bytes + current character
       *
       * 로 판단하고 현재 문자에 resync.
       */

      if (
        ahead &&
        (
          !nextExpected ||
          ahead.position <
          nextExpected.match.position
        )
      ) {

        map.push({

          Character:
            character,

          Status:
            ahead.status,

          Representation:
            ahead.representation,

          Position:
            ahead.position,

          Skipped:
            ahead.skipped,

          Resynced:
            true
        });


        offset =
          ahead.position +
          ahead.consumed;


        resyncCount++;


        continue;
      }


      /*
       * 다음 기대 문자가 먼저 발견되면
       * 현재 문자는 응답 sequence에서 사라진 것으로 본다.
       *
       * REMOVED라고 확정하지 않고 MISSING으로만 기록한다.
       */

      if (
        nextExpected
      ) {

        map.push({

          Character:
            character,

          Status:
            'MISSING',

          Representation:
            '',

          Position:
            offset,

          Skipped:
            nextExpected.match.position -
            offset,

          Resynced:
            true
        });


        /*
         * offset은 건드리지 않는다.
         *
         * 다음 loop에서 next character가 현재 offset 이후에서
         * 다시 정상적으로 탐색된다.
         */

        resyncCount++;


        continue;
      }


      // ----------------------------------------------------------
      // 4. 이후 기대 문자도 발견되지 않음
      // ----------------------------------------------------------

      map.push({

        Character:
          character,

        Status:
          'MISSING_OR_TRUNCATED',

        Representation:
          segment.substring(
            offset,
            offset + 20
          ),

        Position:
          offset,

        Skipped:
          0,

        Resynced:
          false
      });
    }


    return {

      markerPosition,

      reflectedSegment:
        segment.substring(
          0,
          Math.min(
            segment.length,
            150
          )
        ),

      resyncCount,

      charMap:
        map
    };
  }


  // ============================================================
  // AGGREGATE CHARACTER MAP
  // ============================================================

  function aggregateCharMap(
    reflectionAnalyses
  ) {

    const aggregate = {};


    for (
      const character
      of CONFIG.SPECIAL
    ) {

      aggregate[character] = {

        RAW:
          0,

        HTML_ENTITY:
          0,

        JSON_ESCAPE:
          0,

        JS_ESCAPE:
          0,

        URL_ENCODED:
          0,

        MISSING:
          0,

        MISSING_OR_TRUNCATED:
          0
      };
    }


    for (
      const analysis
      of reflectionAnalyses
    ) {

      for (
        const item
        of analysis.charMap
      ) {

        if (
          aggregate[item.Character] &&
          aggregate[item.Character][item.Status] !== undefined
        ) {

          aggregate[item.Character][item.Status]++;
        }
      }
    }


    return aggregate;
  }


  // ============================================================
  // TRANSFORMATION CLASS
  // ============================================================

  function classifyTransformation(
    reflectionAnalyses
  ) {

    if (
      !reflectionAnalyses.length
    ) {

      return 'NOT_REFLECTED';
    }


    const statuses =
      reflectionAnalyses.flatMap(
        analysis =>
          analysis.charMap.map(
            item =>
              item.Status
          )
      );


    if (
      statuses.length &&
      statuses.every(
        status =>
          status === 'RAW'
      )
    ) {

      return 'RAW';
    }


    const rawCount =
      statuses.filter(
        status =>
          status === 'RAW'
      ).length;


    const encodedStatuses =
      statuses.filter(
        status =>
          status === 'HTML_ENTITY' ||
          status === 'JSON_ESCAPE' ||
          status === 'JS_ESCAPE' ||
          status === 'URL_ENCODED'
      );


    const missingStatuses =
      statuses.filter(
        status =>
          status === 'MISSING' ||
          status === 'MISSING_OR_TRUNCATED'
      );


    if (
      encodedStatuses.length &&
      !rawCount &&
      !missingStatuses.length
    ) {

      const encodingTypes =
        new Set(
          encodedStatuses
        );


      if (
        encodingTypes.size === 1
      ) {

        return [
          ...encodingTypes
        ][0];
      }


      return 'FULL_ENCODED_MIXED';
    }


    if (
      encodedStatuses.length &&
      rawCount &&
      !missingStatuses.length
    ) {

      return 'PARTIAL_ENCODED';
    }


    if (
      missingStatuses.length &&
      !rawCount &&
      !encodedStatuses.length
    ) {

      return 'FILTERED_OR_TRUNCATED';
    }


    if (
      missingStatuses.length &&
      (
        rawCount ||
        encodedStatuses.length
      )
    ) {

      return 'PARTIAL_FILTERED';
    }


    return 'OTHER_TRANSFORM';
  }


  // ============================================================
  // CHARACTER SUMMARY
  // ============================================================

  function getCharacterSummary(
    reflectionAnalyses
  ) {

    if (
      !reflectionAnalyses.length
    ) {

      return '-';
    }


    const first =
      reflectionAnalyses[0];


    return first.charMap
      .map(
        item => {

          const suffix =
            item.Resynced
              ? `@+${item.Skipped}`
              : '';

          return (
            `${item.Character}:${item.Status}${suffix}`
          );
        }
      )
      .join(' | ');
  }


  // ============================================================
  // HTML CONTEXT
  // ============================================================

  function classifyHtmlContext(
    body,
    marker
  ) {

    const index =
      body.indexOf(marker);


    if (
      index === -1
    ) {

      return 'NOT_REFLECTED';
    }


    const before =
      body.substring(
        0,
        index
      );


    const lower =
      before.toLowerCase();


    const scriptStart =
      lower.lastIndexOf('<script');

    if (
      scriptStart >
      lower.lastIndexOf('</script')
    ) {

      const tagEnd =
        lower.indexOf('>', scriptStart);

      const openingTag =
        tagEnd === -1
          ? lower.substring(scriptStart)
          : lower.substring(scriptStart, tagEnd + 1);

      if (
        /type\s*=\s*["']?(?:application\/(?:ld\+)?json|importmap|speculationrules)/i.test(
          openingTag
        )
      ) {

        return 'SCRIPT_DATA';
      }

      return 'SCRIPT';
    }


    if (
      lower.lastIndexOf('<style') >
      lower.lastIndexOf('</style')
    ) {

      return 'STYLE';
    }


    const lastLT =
      before.lastIndexOf('<');


    const lastGT =
      before.lastIndexOf('>');


    if (
      lastLT >
      lastGT
    ) {

      const fragment =
        before.substring(
          lastLT
        );


      const dq =
        (
          fragment.match(/"/g) ||
          []
        ).length;


      const sq =
        (
          fragment.match(/'/g) ||
          []
        ).length;


      if (
        dq % 2 === 1
      ) {

        return 'HTML_ATTRIBUTE_DQ';
      }


      if (
        sq % 2 === 1
      ) {

        return 'HTML_ATTRIBUTE_SQ';
      }


      return 'HTML_ATTRIBUTE_UNQUOTED';
    }


    return 'HTML_TEXT';
  }


  // ============================================================
  // CONTEXT CLASSIFIER
  // ============================================================

  function classifyContext(
    body,
    marker,
    contentType
  ) {

    if (
      !body.includes(marker)
    ) {

      return 'NOT_REFLECTED';
    }


    const type =
      normalizeContentType(
        contentType
      );


    if (
      type.includes('json') ||
      type.includes('+json')
    ) {

      return 'JSON';
    }


    if (
      type.includes('javascript') ||
      type.includes('ecmascript')
    ) {

      return 'SCRIPT';
    }


    if (
      type.includes('html') ||
      type.includes('xhtml')
    ) {

      return classifyHtmlContext(
        body,
        marker
      );
    }


    if (
      type.startsWith('text/')
    ) {

      return 'TEXT';
    }


    return 'UNKNOWN';
  }


  // ============================================================
  // VALUE TYPE
  // ============================================================

  function inferValueType(value) {

    if (
      value === ''
    ) {

      return 'EMPTY';
    }


    if (
      /^(?:true|false)$/i.test(value)
    ) {

      return 'BOOLEAN';
    }


    if (
      /^-?(?:\d+|\d*\.\d+)$/.test(value)
    ) {

      return 'NUMBER';
    }


    return 'STRING';
  }


  // ============================================================
  // LENGTH ANALYSIS
  // ============================================================

  function getLengthAnalysis(
    baselineLength,
    testedLength
  ) {

    const diff =
      testedLength -
      baselineLength;


    const absoluteDiff =
      Math.abs(
        diff
      );


    const ratio =
      baselineLength > 0
        ? absoluteDiff /
          baselineLength
        : (
            absoluteDiff > 0
              ? 1
              : 0
          );


    const significant =
      absoluteDiff >
        CONFIG.LENGTH_DIFF_THRESHOLD &&

      ratio >
        CONFIG.LENGTH_DIFF_RATIO_THRESHOLD;


    return {

      diff,

      ratio,

      significant
    };
  }


  // ============================================================
  // RESULT CLASSIFIER
  // ============================================================

  function classifyResult({

    baseline,

    tested,

    reflected,

    exact,

    lengthAnalysis

  }) {

    const statusChanged =
      baseline.status !==
      tested.status;


    const classChanged =
      baseline.statusClass !==
      tested.statusClass;


    const baselineError =
      isErrorClass(
        baseline.statusClass
      );


    const testedError =
      isErrorClass(
        tested.statusClass
      );


    if (
      reflected
    ) {

      if (
        !baselineError &&
        testedError
      ) {

        return 'ERROR_REFLECTION';
      }


      if (
        baselineError
      ) {

        return 'ERROR_BASELINE_REFLECTION';
      }


      if (
        statusChanged ||
        classChanged
      ) {

        return 'STATUS_CHANGED_REFLECTION';
      }


      if (
        exact
      ) {

        return 'STABLE_EXACT_REFLECTION';
      }


      return 'STABLE_TRANSFORMED_REFLECTION';
    }


    if (
      statusChanged ||
      classChanged
    ) {

      return 'VALIDATION_STATUS_CHANGE';
    }


    if (
      lengthAnalysis.significant
    ) {

      return 'RESPONSE_CHANGED';
    }


    return 'NO_REFLECTION';
  }


  // ============================================================
  // PARAMETER MINING (히든 파라미터 탐지 -- Param Miner 방식)
  // ============================================================
  //
  // 묶음(chunk) 안의 후보 이름마다 전부 다른 xsstest 마커+SPECIAL 페이로드를
  // 동시에 실어 요청 1번으로 보낸다. 임의값 하나를 공유해서 "뭔가 영향은
  // 있다"만 먼저 확인하고 나중에 진짜 XSS 마커로 또 테스트하는 2단계 방식은
  // 반사가 확인된 이름마다 요청이 2배로 들었다 -- 처음부터 실제 테스트
  // 페이로드를 쓰면 반사가 보이는 순간 그 자리에서 컨텍스트/문자별 변환까지
  // 포함한 완전한 결과가 나오므로 추가 요청이 필요 없다. 묶음 중 아무 마커도
  // 안 나타났는데 응답은 기준과 다르면(반사 없이 영향만 주는 숨은 파라미터)
  // 그 때만 이분탐색으로 범위를 좁힌다 -- 이 경로는 마커로 직접 증명이 안 되는
  // 경우라 어차피 PARAMETER LOOP에서 한 번 더 확인해야 한다.

  function buildMineProbeURL(baseOriginalURL, paramPayloads) {

    const url = new URL(baseOriginalURL);

    for (const [name, payload] of paramPayloads) {
      url.searchParams.set(name, payload);
    }

    return url;
  }


  function mineProbeDiffersFromBaseline(baseline, probe) {

    if (probe.status !== baseline.status) return true;
    if (probe.statusClass !== baseline.statusClass) return true;
    if (probe.normalizedContentType !== baseline.normalizedContentType) return true;

    return getLengthAnalysis(baseline.length, probe.length).significant;
  }


  function buildMineHitResult({ target, baseline, tested, parameter, marker, payload, testURL }) {

    const markerPositions = findMarkerPositions(tested.body, marker);
    const reflectionAnalyses = markerPositions.map(position => analyzeReflectionAt(tested.body, marker, position));
    const transformationType = classifyTransformation(reflectionAnalyses);
    const charMap = aggregateCharMap(reflectionAnalyses);
    const characterSummary = getCharacterSummary(reflectionAnalyses);
    const resyncCount = reflectionAnalyses.reduce((sum, analysis) => sum + analysis.resyncCount, 0);
    const contextType = classifyContext(tested.body, marker, tested.contentType);
    const contentTypeTransition = getContentTypeTransition(baseline.contentType, tested.contentType);
    const lengthAnalysis = getLengthAnalysis(baseline.length, tested.length);
    const exact = tested.body.includes(payload);
    const resultType = classifyResult({ baseline, tested, reflected: true, exact, lengthAnalysis });

    // testURL은 이 묶음의 다른 후보 이름들까지 전부 같이 실려 나간 실제 요청이라
    // URL 필드에 그대로 쓰면 "파라미터 하나 확인했는데 20개가 전부 xsstest로
    // 찍힌 URL"처럼 보여 오해를 산다. 사람이 수동으로 재현/보고할 때 쓸 URL은
    // 이 파라미터 하나만 건드린 깨끗한 버전으로 따로 만든다 -- 실제 탐지에 쓰인
    // 원요청은 MiningBatchURL로 남겨 필요하면 참고할 수 있게 한다.
    const reproURL = new URL(target.originalURL);
    reproURL.searchParams.set(parameter, payload);

    return {
      Source: target.source,
      Origin: target.sameOrigin ? 'SAME' : 'CROSS',
      ObservedMethod: target.observedMethod,
      ReplayMethod: 'GET',
      Host: target.url.hostname,
      Path: target.url.pathname,
      Parameter: parameter,
      ParamSource: 'PARAM_MINED',
      Payload: payload,
      Original: '',
      OriginalType: 'EMPTY',
      HTTP: `${baseline.status}→${tested.status}`,
      BaselineClass: baseline.statusClass,
      ResponseClass: tested.statusClass,
      StatusChanged: tested.status !== baseline.status,
      StatusClassChanged: tested.statusClass !== baseline.statusClass,
      BaselineContentType: baseline.normalizedContentType,
      ResponseContentType: tested.normalizedContentType,
      ContentTypeChanged: contentTypeTransition !== 'STABLE',
      ContentTypeTransition: contentTypeTransition,
      Reflected: true,
      Occurrences: markerPositions.length,
      Exact: exact,
      Transformation: transformationType,
      CharacterMap: charMap,
      CharacterSummary: characterSummary,
      ResyncCount: resyncCount,
      ReflectionAnalyses: reflectionAnalyses,
      ContextType: contextType,
      BaselineLength: baseline.length,
      ResponseLength: tested.length,
      LengthDiff: lengthAnalysis.diff,
      LengthDiffRatio: Number(lengthAnalysis.ratio.toFixed(4)),
      SignificantLengthChange: lengthAnalysis.significant,
      Result: resultType,
      Context: getContext(tested.body, marker),
      URL: reproURL.href,
      MiningBatchURL: testURL.href,
    };
  }

  /*
   * paramNames 묶음에 이름마다 다른 xsstest 마커를 동시에 실어 요청 1번으로
   * 보낸다. 마커가 나타난 이름은 그 자리에서 바로 완전한 결과를 만들어
   * confirmedResults에 쌓는다(추가 요청 없음, PARAMETER LOOP에서 다시
   * 테스트하지 않음). 아무 마커도 안 나타났는데 응답이 기준과 다르면 반사
   * 없이 영향만 주는 이름이 섞여있다는 뜻이라 그 때만 반으로 쪼개 재귀 --
   * 이렇게 좁혀진 이름은 "후보"로만 반환하고 실제 확인은 PARAMETER LOOP에
   * 맡긴다(마커 직접 증거가 없는 경우라서). budget은 대상 하나당 전체 마이닝
   * 요청 수 상한을 모든 재귀 호출이 공유하는 카운터다.
   */
  async function mineParamChunk(target, baseline, paramNames, budget, confirmedResults) {

    if (!paramNames.length || budget.remaining <= 0) {
      return [];
    }

    budget.remaining--;

    const markerByName = new Map();

    for (const name of paramNames) {
      markerCounter++;
      const marker = `xsstest${markerCounter}`;
      markerByName.set(name, { marker, payload: marker + CONFIG.SPECIAL });
    }

    const testURL = buildMineProbeURL(
      target.originalURL,
      [...markerByName].map(([name, info]) => [name, info.payload])
    );

    let tested;

    try {

      const response = await fetchGET(testURL.href);
      tested = await readResponse(response);

    } catch {

      // 네트워크 오류/타임아웃 -- 이 묶음은 판단 불가로 보고 건너뜀
      return [];
    }

    const reflectedNames = [];

    for (const [name, { marker, payload }] of markerByName) {
      if (tested.body.includes(marker)) {
        reflectedNames.push(name);
        confirmedResults.push(
          buildMineHitResult({ target, baseline, tested, parameter: name, marker, payload, testURL })
        );
      }
    }

    if (reflectedNames.length) {
      // 반사 확인된 이름은 이미 완전한 결과로 등록됐다. 같은 묶음에 반사
      // 없이 영향만 주는 이름이 "동시에" 더 섞여있는 극히 드문 경우까지는
      // 추가 요청을 들여 가리지 않는다.
      return [];
    }

    if (!mineProbeDiffersFromBaseline(baseline, tested)) {
      return [];
    }

    if (paramNames.length === 1) {
      return paramNames;
    }

    const mid = Math.ceil(paramNames.length / 2);

    const left = await mineParamChunk(target, baseline, paramNames.slice(0, mid), budget, confirmedResults);
    const right = await mineParamChunk(target, baseline, paramNames.slice(mid), budget, confirmedResults);

    return [...left, ...right];
  }


  async function mineHiddenParameters(target, baseline, confirmedResults) {

    if (!CONFIG.PARAM_MINING_ENABLED) {
      return [];
    }

    const known = new Set(target.params.map(name => name.toLowerCase()));

    const candidates = CONFIG.PARAM_MINE_WORDLIST.filter(name => !known.has(name.toLowerCase()));

    if (!candidates.length) {
      return [];
    }

    const budget = { remaining: CONFIG.PARAM_MINE_MAX_REQUESTS };

    const candidateNames = new Set();

    for (let i = 0; i < candidates.length; i += CONFIG.PARAM_MINE_CHUNK_SIZE) {

      if (budget.remaining <= 0) {

        console.warn(
          '[히든 파라미터 마이닝]', 'PARAM_MINE_MAX_REQUESTS 상한 도달 -- 이 대상의 나머지 후보는 건너뜀', target.url.pathname
        );

        break;
      }

      const chunk = candidates.slice(i, i + CONFIG.PARAM_MINE_CHUNK_SIZE);

      const remaining = await mineParamChunk(target, baseline, chunk, budget, confirmedResults);

      for (const name of remaining) candidateNames.add(name);
    }

    return [...candidateNames];
  }


  // ============================================================
  // DISCOVERY SOURCES
  // ============================================================

  const rawSources = [];


  // CURRENT
  rawSources.push({

    source:
      'CURRENT',

    method:
      'GET',

    methodConfidence:
      'KNOWN',

    initiatorType:
      'navigation',

      location.href

  const navigationEntries =

  const resourceEntries =
    performance.getEntriesByType(

  for (
    const entry

  ) {

    const initiatorType =
      String(
        entry.initiatorType

    if (
      initiatorType !== 'fetch' &&
      initiatorType !== 'xmlhttprequest'

    rawSources.push({

        'RESOURCE',

      method:
        'UNKNOWN',

      methodConfidence:
        'UNKNOWN',

      initiatorType:
        entry.initiatorType,

      name:
        entry.name
    });
  }


  // DOCUMENT LINKS
  // Resource Timing may be unavailable or cleared by the application. Query
  // URLs already present in same-page link/form markup are safe discovery
  // candidates: this step observes them only; GET replay remains gated by the
  // existing same-origin policy below.
  //
  // GET <form> fields are a common reflected-XSS vector (search boxes, filter
  // forms) but a form's own `.action` property NEVER includes its input
  // values -- those only become a query string once the browser actually
  // submits the form. Without synthesizing that query string here, every GET
  // form's parameters were silently invisible to discovery (url.search was
  // always empty, so the whole form got dropped by the `!url.search` filter
  // in the DISCOVERY step below). Also switched the selector from
  // `form[action]` to plain `form`, since a form with no `action` attribute
  // at all (submits to the current page, a very common pattern) has no
  // `action` attribute to match on but is still a perfectly valid GET form.
  const buildFormGetURL = form => {
    const actionAttr = form.getAttribute('action');
    const url = new URL(actionAttr || location.href, location.href);
    const params = new URLSearchParams(url.search);
    form.querySelectorAll('input[name], select[name], textarea[name]').forEach(field => {
      if (field.disabled) return;
      const type = (field.type || '').toLowerCase();
      if (type === 'submit' || type === 'button' || type === 'reset' || type === 'file' || type === 'image') return;
      if ((type === 'checkbox' || type === 'radio') && !field.checked) return;
      params.set(field.name, field.value ?? '');
    });
    url.search = params.toString();
    return url.href;
  };

  for (
    const element
    of document.querySelectorAll(
      'a[href], area[href], form'
    )
  ) {

    if (
      element instanceof HTMLFormElement &&
      element.method.toUpperCase() !== 'GET'
    ) {
      continue;
    }

    const value =
      element instanceof HTMLFormElement
        ? buildFormGetURL(element)
        : element.href;


    if (!value) {
      continue;
    }


    rawSources.push({

      source:
        'DOCUMENT_LINK',

      method:
        'GET',

      methodConfidence:
        'ASSUMED',

      initiatorType:
        element.tagName.toLowerCase(),

      name:
        value
    });
  }


  // ============================================================
  // DISCOVERY
  // ============================================================

  const discovered = [];


  for (
    const source
    of rawSources
  ) {

    try {

      const url =
        new URL(
          source.name
        );


      if (
        CONFIG.STATIC_EXT.test(
          url.pathname
        )
      ) {

        continue;
      }


      if (
        !url.search &&
        !CONFIG.PARAM_MINING_ENABLED
      ) {

        continue;
      }


      const params =
        getParameters(
          url
        );


      if (
        !params.length &&
        !CONFIG.PARAM_MINING_ENABLED
      ) {

        continue;
      }


      const sameOrigin =
        url.origin ===
        location.origin;


      if (
        sameOrigin &&
        !CONFIG.TEST_SAME_ORIGIN
      ) {

        continue;
      }


      if (
        !sameOrigin &&
        !CONFIG.TEST_CROSS_ORIGIN
      ) {

        continue;
      }


      discovered.push({

        source:
          source.source,

        observedMethod:
          source.method,

        methodConfidence:
          source.methodConfidence,

        originalURL:
          url.href,

        url,

        params,

        sameOrigin,

        initiatorType:
          source.initiatorType
      });


    } catch {

      // malformed URL

    }
  }


  // ============================================================
  // DEDUP
  // ============================================================

  const priority = {

    CURRENT:
      3,

    NAVIGATION:
      2,

    DOCUMENT_LINK:
      2,

    RESOURCE:
      1
  };


  const uniqueMap =
    new Map();


  for (
    const item
    of discovered
  ) {

    const parameterStructure =
      [...item.params]
        .sort()
        .join('&');


    const key =
      [
        item.url.origin,
        item.url.pathname,
        parameterStructure
      ].join('|');


    const previous =
      uniqueMap.get(
        key
      );


    if (
      !previous ||
      priority[item.source] >
      priority[previous.source]
    ) {

      uniqueMap.set(
        key,
        item
      );
    }
  }


  const targets =
    [
      ...uniqueMap.values()
    ];


  // ============================================================
  // DISCOVERY OUTPUT
  // ============================================================

  console.log('');

  console.log(
    '================ XSS Scout -- GET 파라미터 반사 스캔 시작 ================'
  );


  console.log(
    `[원본 수집 소스] ${rawSources.length}개`
  );


  console.log(
    `[테스트 대상 후보] ${targets.length}개 (경로+파라미터 조합, 중복 제거됨)`
  );


  targets.forEach((target, index) => {
    console.log(
      `#${index + 1} [${target.source}/${target.observedMethod}/${target.sameOrigin ? 'SAME' : 'CROSS'}] ${target.url.hostname}${target.url.pathname}  파라미터=${target.params.join(', ') || '-'}  발견경로=${target.initiatorType}`
    );
  });


  // ============================================================
  // RESULT ARRAYS
  // ============================================================

  const results = [];

  const replayableTargets = [];

  const unreplayableTargets = [];

  const staticTargets = [];

  const minedParameterFindings = [];

  let minedParameterTotal = 0;

  let minedConfirmedTotal = 0;

  let noParamMiningTargetsUsed = 0;

  let markerCounter = 0;


  // ============================================================
  // TARGET LOOP
  // ============================================================

  for (
    let targetIndex = 0;
    targetIndex < targets.length;
    targetIndex++
  ) {

    const target =
      targets[targetIndex];


    console.log('');

    console.log(
      '================================================'
    );


    console.log(
      `[대상 ${targetIndex + 1}/${targets.length}]`,
      `[${target.source}]`,
      target.url.pathname
    );


    // ==========================================================
    // BASELINE
    // ==========================================================

    let baseline;


    try {

      console.log(
        '[기준 요청(GET)]',
        target.originalURL
      );


      const response =
        await fetchGET(
          target.originalURL
        );


      baseline =
        await readResponse(
          response
        );


      console.log(
        '[기준 응답]',
        `HTTP=${baseline.status}`,
        `클래스=${baseline.statusClass}`,
        `길이=${baseline.length}`,
        `Type=${baseline.normalizedContentType}`
      );


    } catch (error) {

      unreplayableTargets.push({

        수집출처:
          target.source,

        '출처(동일/교차)':
          target.sameOrigin
            ? 'SAME'
            : 'CROSS',

        호스트:
          target.url.hostname,

        경로:
          target.url.pathname,

        관측된메서드:
          target.observedMethod,

        probe메서드:
          'GET',

        사유:
          '네트워크 오류 또는 CORS 차단',

        오류내용:
          String(error)
      });


      console.warn(
        '[재현 불가]',
        '네트워크 오류 또는 CORS 차단',
        target.url.pathname
      );


      continue;
    }


    // ==========================================================
    // METHOD REJECT
    // ==========================================================

    if (
      CONFIG.METHOD_REJECT_STATUS.has(
        baseline.status
      )
    ) {

      unreplayableTargets.push({

        수집출처:
          target.source,

        '출처(동일/교차)':
          target.sameOrigin
            ? 'SAME'
            : 'CROSS',

        호스트:
          target.url.hostname,

        경로:
          target.url.pathname,

        관측된메서드:
          target.observedMethod,

        probe메서드:
          'GET',

        HTTP:
          baseline.status,

        기준클래스:
          baseline.statusClass,

        사유:
          '서버가 메서드를 거부함(405/501)'
      });


      console.warn(
        '[재현 불가]',
        `메서드 거부됨 HTTP=${baseline.status}`,
        target.url.pathname
      );


      continue;
    }


    // ==========================================================
    // STATIC
    // ==========================================================

    if (
      CONFIG.SKIP_CONTENT_TYPE.test(
        baseline.contentType
      )
    ) {

      staticTargets.push({

        수집출처:
          target.source,

        호스트:
          target.url.hostname,

        경로:
          target.url.pathname,

        HTTP:
          baseline.status,

        기준클래스:
          baseline.statusClass,

        'Content-Type':
          baseline.normalizedContentType,

        사유:
          '정적 리소스 응답(CSS/이미지/폰트 등)'
      });


      console.log(
        '[정적 응답]',
        baseline.normalizedContentType,
        target.url.pathname
      );


      continue;
    }


    // ==========================================================
    // PARAMETER MINING (히든 파라미터 탐지 -- Param Miner 방식)
    // ==========================================================

    const isNoParamTarget =
      target.params.length === 0;


    const noParamMiningBudgetLeft =
      !isNoParamTarget ||
      noParamMiningTargetsUsed < CONFIG.PARAM_MINE_MAX_NO_PARAM_TARGETS;


    if (
      isNoParamTarget &&
      !noParamMiningBudgetLeft
    ) {

      console.warn(
        '[히든 파라미터 마이닝]',
        'PARAM_MINE_MAX_NO_PARAM_TARGETS 상한 도달 -- 보이는 파라미터 없는 이 대상은 마이닝 건너뜀',
        target.url.pathname
      );
    }


    const minedConfirmedResults =
      [];


    const minedParams =
      noParamMiningBudgetLeft
        ? await mineHiddenParameters(
            target,
            baseline,
            minedConfirmedResults
          )
        : [];


    if (
      minedConfirmedResults.length
    ) {

      minedConfirmedTotal +=
        minedConfirmedResults.length;

      results.push(
        ...minedConfirmedResults
      );

      console.log(
        '[히든 파라미터 확인됨 -- 반사됨]',
        minedConfirmedResults.map(result => result.Parameter).join(', '),
        '--',
        target.url.pathname,
        '(마이닝 단계에서 이미 완전히 테스트됨, 추가 요청 없음)'
      );
    }


    if (
      isNoParamTarget &&
      noParamMiningBudgetLeft
    ) {

      noParamMiningTargetsUsed++;
    }


    if (
      minedParams.length
    ) {

      minedParameterTotal +=
        minedParams.length;


      for (
        const name
        of minedParams
      ) {

        minedParameterFindings.push({

          호스트:
            target.url.hostname,

          경로:
            target.url.pathname,

          숨은파라미터후보:
            name,

          사유:
            '마커 반사는 없었지만 응답이 기준과 다름(상태/타입/길이) -- 반사 없이 영향만 주는 유형이라 확정 아님',

          기준URL:
            target.originalURL
        });
      }


      console.log(
        '[히든 파라미터 후보 -- 반사 미확인, 추가 확인 필요]',
        minedParams.join(', '),
        '--',
        target.url.pathname
      );
    }


    const allParams =
      [
        ...target.params,
        ...minedParams
      ];


    // ==========================================================
    // REPLAYABLE
    // ==========================================================

    replayableTargets.push({

      수집출처:
        target.source,

      '출처(동일/교차)':
        target.sameOrigin
          ? 'SAME'
          : 'CROSS',

      호스트:
        target.url.hostname,

      경로:
        target.url.pathname,

      관측된메서드:
        target.observedMethod,

      재현메서드:
        'GET',

      기준HTTP:
        baseline.status,

      기준클래스:
        baseline.statusClass,

      'Content-Type':
        baseline.normalizedContentType,

      파라미터목록:
        target.params.join(', '),

      히든파라미터후보목록:
        minedParams.join(', ') || '-'
    });


    // ==========================================================
    // PARAMETER LOOP
    // ==========================================================

    for (
      const parameter
      of allParams
    ) {

      markerCounter++;


      const marker =
        `xsstest${markerCounter}`;


      const payload =
        marker +
        CONFIG.SPECIAL;


      const testURL =
        new URL(
          target.originalURL
        );


      const originalValues =
        getOriginalValues(
          testURL,
          parameter
        );


      const originalValue =
        originalValues.length
          ? originalValues[0]
          : '';


      const valueType =
        inferValueType(
          originalValue
        );


      testURL.searchParams.set(
        parameter,
        payload
      );


      console.log('');

      console.log(
        `[테스트 ${markerCounter}]`,
        `파라미터=${parameter}`,
        `페이로드=${payload}`,
        `원본타입=${valueType}`
      );


      try {

        const response =
          await fetchGET(
            testURL.href
          );


        const tested =
          await readResponse(
            response
          );


        // --------------------------------------------------------
        // REFLECTION
        // --------------------------------------------------------

        const markerPositions =
          findMarkerPositions(
            tested.body,
            marker
          );


        const reflected =
          markerPositions.length > 0;


        const exact =
          reflected &&
          tested.body.includes(
            payload
          );


        // --------------------------------------------------------
        // CHARACTER-LEVEL TRANSFORMATION
        // --------------------------------------------------------

        const reflectionAnalyses =
          markerPositions.map(
            position =>
              analyzeReflectionAt(
                tested.body,
                marker,
                position
              )
          );


        const transformationType =
          classifyTransformation(
            reflectionAnalyses
          );


        const charMap =
          aggregateCharMap(
            reflectionAnalyses
          );


        const characterSummary =
          getCharacterSummary(
            reflectionAnalyses
          );


        const resyncCount =
          reflectionAnalyses.reduce(
            (
              sum,
              analysis
            ) =>
              sum +
              analysis.resyncCount,
            0
          );


        // --------------------------------------------------------
        // CONTEXT
        // --------------------------------------------------------

        const contextType =
          reflected
            ? classifyContext(
                tested.body,
                marker,
                tested.contentType
              )
            : '-';


        // --------------------------------------------------------
        // RESPONSE STATE
        // --------------------------------------------------------

        const statusChanged =
          tested.status !==
          baseline.status;


        const statusClassChanged =
          tested.statusClass !==
          baseline.statusClass;


        const contentTypeTransition =
          getContentTypeTransition(
            baseline.contentType,
            tested.contentType
          );

          contentTypeTransition !==
          'STABLE';


        const lengthAnalysis =
          getLengthAnalysis(
            baseline.length,
            tested.length
          );


        // --------------------------------------------------------
        // RESULT
        // --------------------------------------------------------

        const resultType =
          classifyResult({

            baseline,

            tested,

            reflected,

            exact,

            lengthAnalysis
          });


        results.push({

          Source:
            target.source,

          Origin:
            target.sameOrigin
              ? 'SAME'
              : 'CROSS',

          ObservedMethod:
            target.observedMethod,

          ReplayMethod:
            'GET',

          Host:
            target.url.hostname,

          Path:
            target.url.pathname,

          Parameter:
            parameter,

          ParamSource:
            target.params.includes(parameter)
              ? 'DISCOVERED'
              : 'PARAM_MINED_CANDIDATE',

          Payload:
            payload,

          Original:
            originalValue,

            valueType,

          HTTP:
            `${baseline.status}→${tested.status}`,

          ResponseClass:
            tested.statusClass,

            exact,

// v2.9 analysis executes after v2.8 result persistence.
(() => {

  // ============================================================
  // XSS SCOUT v2.9
  // PASSIVE RESULT ANALYZER
  //
  // 입력:
  //   window.__XSS_SCOUT__.results
  //
  // 기능:
  //   - 추가 HTTP 요청 없음
  //   - HIGH / MEDIUM / LOW / INFO signal 분류
  //   - endpoint 단위 saturation 계산
  //   - 반복적인 FULL URL encoding noise 압축
  //   - 원본 results 보존
  // ============================================================


  const scout =
    window.__XSS_SCOUT__;


  if (
    !scout ||
    !Array.isArray(scout.results)
  ) {

    console.error(
      '[XSS Scout v2.9] v2.8 results가 없습니다.'
    );

    return;
  }


  const results =
    scout.results;


  // ============================================================
  // HELPERS
  // ============================================================

  function endpointKey(result) {

    return [
      result.Origin || '',
      result.Host || '',
      result.Path || ''
    ].join('|');
  }


  function isErrorReflection(result) {

    return (
      result.Result === 'ERROR_REFLECTION' ||
      result.Result === 'ERROR_BASELINE_REFLECTION'
    );
  }


  function isStableReflection(result) {

    return (
      result.Result === 'STABLE_EXACT_REFLECTION' ||
      result.Result === 'STABLE_TRANSFORMED_REFLECTION'
    );
  }


  function isFilteredTransform(result) {

    return (
      result.Transformation === 'PARTIAL_FILTERED' ||
      result.Transformation === 'FILTERED_OR_TRUNCATED'
    );
  }


  function isFullUrlEncoded(result) {

    const map =
      result.CharacterMap;


    if (!map) {
      return false;
    }


    const special =
      [...`<>;:'"\``];


    let seen = 0;


    for (
      const character
      of special
    ) {

      const entry =
        map[character];


      if (!entry) {
        return false;
      }


      const total =
        Object.values(entry)
          .reduce(
            (sum, value) =>
              sum +
              (
                Number(value) || 0
              ),
            0
          );


      if (
        total === 0
      ) {

        return false;
      }


      if (
        (
          entry.URL_ENCODED || 0
        ) !== total
      ) {

        return false;
      }


      seen++;
    }


    return (
      seen ===
      special.length
    );
  }


  function hasMissingCharacter(result) {

    const map =
      result.CharacterMap;


    if (!map) {
      return false;
    }


    return Object
      .values(map)
      .some(
        entry =>
          (
            (
              entry?.MISSING || 0
            ) +

            (
              entry?.MISSING_OR_TRUNCATED || 0
            )
          ) > 0
      );
  }


  // ============================================================
  // ENDPOINT GROUPING
  // ============================================================

  const endpointMap =
    new Map();


  for (
    const result
    of results
  ) {

    const key =
      endpointKey(
        result
      );


    if (
      !endpointMap.has(key)
    ) {

      endpointMap.set(
        key,
        {
          key,

          Origin:
            result.Origin,

          Host:
            result.Host,

          Path:
            result.Path,

          results:
            []
        }
      );
    }


    endpointMap
      .get(key)
      .results
      .push(result);
  }


  // ============================================================
  // ENDPOINT STATISTICS
  // ============================================================

  const endpointStats = [];


  for (
    const endpoint
    of endpointMap.values()
  ) {

    const rows =
      endpoint.results;


    const tested =
      rows.length;


    const reflected =
      rows.filter(
        row =>
          row.Reflected
      ).length;


    const stable =
      rows.filter(
        isStableReflection
      ).length;


    const errors =
      rows.filter(
        isErrorReflection
      ).length;


    const raw =
      rows.filter(
        row =>
          row.Transformation === 'RAW'
      ).length;


    const partial =
      rows.filter(
        row =>
          row.Transformation === 'PARTIAL_ENCODED'
      ).length;


    const filtered =
      rows.filter(
        isFilteredTransform
      ).length;


    const fullUrlEncoded =
      rows.filter(
        row =>
          row.Reflected &&
          isFullUrlEncoded(row)
      ).length;


    const reflectionRate =
      tested
        ? reflected / tested
        : 0;


    const fullUrlEncodingRate =
      reflected
        ? fullUrlEncoded / reflected
        : 0;


    // ----------------------------------------------------------
    // DOMINANT TRANSFORMATION
    // ----------------------------------------------------------

    const transformCounts =
      new Map();


    for (
      const row
      of rows
    ) {

      if (
        !row.Reflected
      ) {
        continue;
      }


      const type =
        row.Transformation ||
        'UNKNOWN';


      transformCounts.set(
        type,
        (
          transformCounts.get(type) ||
          0
        ) + 1
      );
    }


    const dominantTransform =
      [
        ...transformCounts.entries()
      ]

        .sort(
          (a, b) =>
            b[1] - a[1]
        )[0]?.[0] ||
        '-';


    // ----------------------------------------------------------
    // SATURATION
    // ----------------------------------------------------------

    let saturation =
      'NONE';


    if (
      tested >= 3 &&
      reflectionRate >= 0.90
    ) {

      saturation =
        'VERY_HIGH';

    } else if (
      tested >= 3 &&
      reflectionRate >= 0.60
    ) {

      saturation =
        'HIGH';

    } else if (
      reflectionRate > 0
    ) {

      saturation =
        'PARTIAL';
    }


    // ----------------------------------------------------------
    // HOMOGENEOUS URL ENCODING
    // ----------------------------------------------------------

    const homogeneousUrlEncoding =
      (
        reflected >= 3 &&
        fullUrlEncodingRate >= 0.90
      );


    endpointStats.push({

      Key:
        endpoint.key,

      Origin:
        endpoint.Origin,

      Host:
        endpoint.Host,

      Path:
        endpoint.Path,

      Tested:
        tested,

      Reflected:
        reflected,

      ReflectionRate:
        Number(
          reflectionRate
            .toFixed(3)
        ),

      Stable:
        stable,

      ErrorReflection:
        errors,

      Raw:
        raw,

      PartialEncoded:
        partial,

      Filtered:
        filtered,

      FullUrlEncoded:
        fullUrlEncoded,

      FullUrlEncodingRate:
        Number(
          fullUrlEncodingRate
            .toFixed(3)
        ),

      DominantTransform:
        dominantTransform,

      Saturation:
        saturation,

      HomogeneousUrlEncoding:
        homogeneousUrlEncoding
    });
  }


  // ============================================================
  // LOOKUP
  // ============================================================

  const endpointLookup =
    new Map(
      endpointStats.map(
        stat => [
          stat.Key,
          stat
        ]
      )
    );


  // ============================================================
  // SIGNAL CLASSIFICATION
  // ============================================================

  function classifySignal(result) {

    if (
      !result.Reflected
    ) {

      if (
        result.Result ===
        'VALIDATION_STATUS_CHANGE'
      ) {

        return {
          Signal:
            'INFO',

          Reason:
            'VALIDATION_CHANGE'
        };
      }


      if (
        result.Result ===
        'RESPONSE_CHANGED'
      ) {

        return {
          Signal:
            'INFO',

          Reason:
            'RESPONSE_CHANGE'
        };
      }


      return {
        Signal:
          'INFO',

        Reason:
          'NO_REFLECTION'
      };
    }


    // ----------------------------------------------------------
    // ERROR REFLECTION
    // ----------------------------------------------------------

    if (
      isErrorReflection(
        result
      )
    ) {

      return {
        Signal:
          'LOW',

        Reason:
          'ERROR_RESPONSE_REFLECTION'
      };
    }


    // ----------------------------------------------------------
    // NON-EXECUTABLE SCRIPT DATA
    //
    // application/json, import maps, and speculation rules are serialized
    // data containers, not executable JavaScript. Preserve the reflection as
    // a reviewable fact without promoting it to an XSS finding.
    // ----------------------------------------------------------

    if (
      result.ContextType ===
      'SCRIPT_DATA'
    ) {

      return {
        Signal:
          'LOW',

        Reason:
          'SERIALIZED_SCRIPT_DATA_REFLECTION'
      };
    }


    // ----------------------------------------------------------
    // SERIALIZED JSON
    //
    // A response declared as JSON is data, not an HTML execution context.
    // Keep it visible for data-flow review, but do not promote reflection in
    // the response itself to a direct XSS finding.
    // ----------------------------------------------------------

    if (
      result.ContextType ===
      'JSON'
    ) {

      return {
        Signal:
          'LOW',

        Reason:
          'SERIALIZED_JSON_REFLECTION'
      };
    }


    // ----------------------------------------------------------
    // SCRIPT: 문자 단위 검사만으론 실제 JS 문자열을 벗어났는지 확인 불가
    //
    // (피드백 반영) 이전엔 ", ', < 중 하나가 raw로 살아남으면 HIGH를 줬는데
    // 과한 판정이었다 -- 반사된 따옴표가 실제로 감싸는 문자열의 구분자와
    // 같은 종류인지, 주석/정규식/템플릿 리터럴 안인지는 raw 문자 존재
    // 여부만으론 모른다(AST 파싱 없이는 확인 불가). "SCRIPT"는 직렬화된
    // 부트스트랩 데이터도 포함하므로 더더욱 실행 가능한 JS라고 단정할 수
    // 없다 -- raw 구분자 생존 여부와 무관하게 항상 "문법 확인 후보"
    // (MEDIUM)로만 분류한다.
    // ----------------------------------------------------------

    if (result.ContextType === 'SCRIPT') {

      return {
        Signal:
          'MEDIUM',

        Reason:
          'SCRIPT_SYNTAX_REVIEW'
      };
    }

    if (
      result.ContextType === 'TEXT' ||
      result.ContextType === 'UNKNOWN' ||
      result.ContextType === 'STYLE'
    ) {

      // HTML 파싱 컨텍스트가 아니다(text/plain 응답, <style> 안 CSS 등) --
      // raw로 반사돼도 그 자체로는 실행되지 않는다.
      return {
        Signal:
          'LOW',

        Reason:
          'NON_HTML_OR_DATA_REFLECTION'
      };
    }


    // ----------------------------------------------------------
    // 컨텍스트별 정밀 판정: 이 반사 위치를 실제로 벗어나는 데 필요한 구분자
    // 문자 하나가 raw로 살아남았는지만 본다. HTML_TEXT의 '<'와 속성의
    // 매칭 따옴표, 이 둘만 문자 단위 관찰로도 파서 레벨 컨텍스트 이탈을
    // 확신할 수 있는 경우라서 여기서만 HIGH를 준다 -- 그 외(SCRIPT 등)는
    // 위에서 이미 갈렸다.
    //
    // 구분자가 raw로 안 남았어도 다른 문자가 필터링/제거(MISSING)된 흔적이
    // 있으면 "여기선 못 깼지만 서버가 입력을 거르고 있다"는 정황 증거라
    // LOW 대신 MEDIUM까지 올린다 -- 확정 증거는 아니라서 HIGH는 아니다
    // (이전엔 필터링 흔적만으로 HIGH를 줬는데, SCRIPT 판정과 같은 이유로
    // 같이 낮췄다: 문자 단위 관찰만으론 실제 실행 가능 여부를 확정 못 한다).
    // ----------------------------------------------------------

    const rawChar = character => (result.CharacterMap?.[character]?.RAW || 0) > 0;

    const filteringObserved =
      isFilteredTransform(result) ||
      hasMissingCharacter(result);

    if (result.ContextType === 'HTML_TEXT') {

      if (rawChar('<')) {
        return { Signal: 'HIGH', Reason: 'RAW_HTML_TAG_DELIMITER' };
      }

      return filteringObserved
        ? { Signal: 'MEDIUM', Reason: 'CHARACTER_FILTERING' }
        : { Signal: 'LOW', Reason: 'HTML_TEXT_TAG_DELIMITER_ENCODED' };
    }

    if (
      result.ContextType === 'HTML_ATTRIBUTE_DQ' ||
      result.ContextType === 'HTML_ATTRIBUTE_SQ'
    ) {

      const delimiter = result.ContextType === 'HTML_ATTRIBUTE_DQ' ? '"' : "'";

      if (rawChar(delimiter)) {
        return { Signal: 'HIGH', Reason: 'RAW_ATTRIBUTE_DELIMITER' };
      }

      return filteringObserved
        ? { Signal: 'MEDIUM', Reason: 'CHARACTER_FILTERING' }
        : { Signal: 'LOW', Reason: 'ATTRIBUTE_DELIMITER_ENCODED' };
    }

    // HTML_ATTRIBUTE_UNQUOTED: 공백 하나로도 새 속성을 주입해 깰 수 있는
    // 컨텍스트라 SPECIAL 문자셋만으론 판단 불가 -- 항상 수동 검토 대상.
    return {
      Signal:
        'MEDIUM',

      Reason:
        'UNQUOTED_ATTRIBUTE_REVIEW'
    };
  }


  // ============================================================
  // ENRICH RESULTS
  // ============================================================

  const analyzedResults =
    results.map(
      result => {

        const signal =
          classifySignal(
            result
          );


        const endpoint =
          endpointLookup.get(
            endpointKey(result)
          );


        return {

          ...result,

          Signal:
            signal.Signal,

          SignalReason:
            signal.Reason,

          EndpointReflectionRate:
            endpoint
              ?.ReflectionRate ??
            0,

          EndpointSaturation:
            endpoint
              ?.Saturation ??
            'NONE',

          EndpointDominantTransform:
            endpoint
              ?.DominantTransform ??
            '-'
        };
      }
    );


  // ============================================================
  // PRIORITY ORDER
  // ============================================================

  const signalWeight = {

    HIGH:
      4,

    MEDIUM:
      3,

    LOW:
      2,

    INFO:
      1
  };


  const priorityResults =
    [...analyzedResults]

      .sort(
        (a, b) => {

          const signalDiff =
            (
              signalWeight[b.Signal] ||
              0
            ) -
            (
              signalWeight[a.Signal] ||
              0
            );


          if (
            signalDiff !== 0
          ) {

            return signalDiff;
          }


          if (
            a.Reflected !==
            b.Reflected
          ) {

            return Number(
              b.Reflected
            ) -
            Number(
              a.Reflected
            );
          }


          return (
            (
              b.EndpointReflectionRate ||
              0
            ) -
            (
              a.EndpointReflectionRate ||
              0
            )
          );
        }
      );


  // ============================================================
  // GROUPS
  // ============================================================

  const highSignal =
    priorityResults.filter(
      row =>
        row.Signal === 'HIGH'
    );


  const mediumSignal =
    priorityResults.filter(
      row =>
        row.Signal === 'MEDIUM'
    );


  const lowSignal =
    priorityResults.filter(
      row =>
        row.Signal === 'LOW'
    );


  const infoSignal =
    priorityResults.filter(
      row =>
        row.Signal === 'INFO'
    );


  // ============================================================
  // ENDPOINT SUMMARY
  // ============================================================

  console.log('');

  console.log(
    '================ XSS Scout -- 결과 분석(위험도 분류) ================'
  );


  // ============================================================
  // GLOSSARY -- 아래부터 찍히는 용어 전부 여기서 미리 설명. 콘솔 글만
  // 보고도 뜻을 알아야 하니 결과 찍기 전에 한 번 먼저 보여준다.
  // ============================================================

  console.log('');
  console.log('================ 용어 설명 (아래 결과 읽기 전에 먼저 볼 것) ================');
  console.log('');
  console.log('[위험도] HIGH=문자 하나(< 또는 "/\') 수준에서 파싱 구조가 실제로 깨졌다고 확인된 것, 가장 먼저 검증.  MEDIUM=반사는 됐는데 문자 하나만 봐서는 구조가 깨졌는지 확신 못 하는 것(스크립트 안 반사, 따옴표 없는 속성 등), 수동 검토 필요.  LOW=반사됐지만 위험도 낮은 것(텍스트/데이터 영역 등).  INFO=반사 자체가 없거나 참고용.');

  console.log('');
  console.log('바로 아래부터 나오는 표/목록은 전부 위 용어로 찍힘. 결과만 빨리 보고 싶으면 맨 아래 "최종 확인: 취약점 가능성" 블록으로 건너뛸 것.');


  console.log('');

  console.log(
    '================ 엔드포인트별 요약 ================'
  );


  const endpointSummarySorted =
    [...endpointStats].sort((a, b) => {
      if (a.HomogeneousUrlEncoding !== b.HomogeneousUrlEncoding) {
        return Number(a.HomogeneousUrlEncoding) - Number(b.HomogeneousUrlEncoding);
      }
      return b.ReflectionRate - a.ReflectionRate;
    });

  for (const endpoint of endpointSummarySorted) {
    console.log(
      `[${endpoint.Origin}] ${endpoint.Host}${endpoint.Path}  테스트=${endpoint.Tested}건  반사=${endpoint.Reflected}건(${endpoint.ReflectionRate})  주요변환=${endpoint.DominantTransform}  포화도=${endpoint.Saturation}  URL인코딩=${endpoint.FullUrlEncoded}건(${endpoint.FullUrlEncodingRate}, 반복적=${endpoint.HomogeneousUrlEncoding})`
    );
  }


  // ============================================================
  // PRIORITY RESULTS
  // ============================================================

  console.log('');

  console.log(

    // 반응을 클릭 없이 바로 읽을 수 있게 전부 텍스트 줄로 찍는다.
    for (const row of highSignal) {

  console.log(Object.entries({
