(async () => {
  /*
   * XSS Scout v3.3 add-on: passive DOM, controlled-form-marker, request-shape,
   * and navigation observer.
   * It records only metadata and masked previews. It does not inject a
   * payload, navigate, alter request headers, or modify application state.
   * Reload the page to remove the temporary observers.
   */
  const __XSS_SCOUT_DOM__ = (() => {
    const MAX_EVENTS = 250;
    const events = [];
    const requestShapes = [];
    const formMarkers = new Set();
    const xhrMeta = new WeakMap();
    const restores = [];
    const hookStatus = Object.create(null);
    const startedAt = Date.now();

    const mask = value => {
      const text = String(value ?? '');
      if (!text) return '';
      return `[masked:${text.length}]`;
    };

    const sourceValues = () => {
      const values = new Set();
      try {
        const u = new URL(location.href);
        for (const value of u.searchParams.values()) if (value) values.add(value);
        if (u.hash.length > 1) values.add(decodeURIComponent(u.hash.slice(1)));
      } catch { /* malformed location values are ignored */ }
      for (const marker of formMarkers) values.add(marker);
      return [...values].filter(value => value.length >= 3 && value.length <= 4096);
    };

    const routeLabel = value => {
      try {
        const u = new URL(value, location.href);
        const names = [...new Set(u.searchParams.keys())];
        return `${u.origin}${u.pathname}${names.length ? `?${names.map(name => `${encodeURIComponent(name)}=[redacted]`).join('&')}` : ''}`;
      } catch { return '[unparseable-url]'; }
    };

    const bodyShape = body => {
      if (body === undefined || body === null || body === '') return 'NONE';
      if (body instanceof FormData) return `FORM_DATA(keys:${[...body.keys()].map(String).slice(0, 20).join(',')})`;
      if (body instanceof URLSearchParams) return `URL_ENCODED(keys:${[...body.keys()].map(String).slice(0, 20).join(',')})`;
      if (typeof body !== 'string') return `NON_TEXT(${body.constructor?.name || typeof body})`;
      try {
        const parsed = JSON.parse(body);
        if (parsed && typeof parsed === 'object') return `JSON(keys:${Object.keys(parsed).slice(0, 20).join(',')})`;
      } catch { /* not JSON */ }
      if (/^[^=&]+=[^&]*(?:&|$)/.test(body)) {
        try { return `URL_ENCODED(keys:${[...new URLSearchParams(body).keys()].slice(0, 20).join(',')})`; } catch { /* fall through */ }
      }
      return `TEXT(length:${body.length})`;
    };

    const addRequestShape = (transport, method, value, body) => {
      if (requestShapes.length >= MAX_EVENTS) return;
      let sameOrigin = false;
      try { sameOrigin = new URL(value, location.href).origin === location.origin; } catch { /* handled by route label */ }
      if (!sameOrigin) return;
      requestShapes.push({
        Time: new Date().toISOString(), Transport: transport,
        Method: String(method || 'GET').toUpperCase(), URL: routeLabel(value),
        BodyShape: bodyShape(body),
      });
    };

    const sourceMatch = value => {
      const text = String(value ?? '');
      const matched = sourceValues().some(source => text.includes(source));
      return {
        matched,
        scoutMarker: /xss(?:scout|test)/i.test(text),
      };
    };

    const add = (kind, detail = {}) => {
      if (events.length >= MAX_EVENTS) return;
      const probe = sourceMatch(detail.value);
      events.push({
        Time: new Date().toISOString(),
        Kind: kind,
        Target: detail.target || '',
        Attribute: detail.attribute || '',
        Value: mask(detail.value),
        SourceMatch: probe.matched,
        ScoutMarker: probe.scoutMarker,
        Origin: detail.origin || '',
        CrossOrigin: Boolean(detail.crossOrigin),
      });
    };

    const patchMethod = (object, name, handler, label = name) => {
      const original = object?.[name];
      if (typeof original !== 'function') {
        hookStatus[label] = 'UNAVAILABLE';
        return false;
      }
      try {
        object[name] = function (...args) {
          try { handler.call(this, args); } catch { /* observation must never break page code */ }
          return original.apply(this, args);
        };
        restores.push(() => { object[name] = original; });
        hookStatus[label] = 'ACTIVE';
        return true;
      } catch {
        hookStatus[label] = 'UNAVAILABLE';
        return false;
      }
    };

    const patchSetter = (prototype, name, kind, label = name) => {
      try {
        const descriptor = Object.getOwnPropertyDescriptor(prototype, name);
        if (!descriptor?.set || !descriptor.configurable) {
          hookStatus[label] = 'UNAVAILABLE';
          return false;
        }
        Object.defineProperty(prototype, name, {
          configurable: true,
          enumerable: descriptor.enumerable,
          get: descriptor.get,
          set(value) {
            add(kind, { target: this?.tagName || prototype.constructor?.name || '', value });
            return descriptor.set.call(this, value);
          },
        });
        restores.push(() => Object.defineProperty(prototype, name, descriptor));
        hookStatus[label] = 'ACTIVE';
        return true;
      } catch {
        hookStatus[label] = 'UNAVAILABLE';
        return false;
      }
    };

    patchSetter(Element.prototype, 'innerHTML', 'HTML_SINK_INNER_HTML', 'Element.innerHTML');
    patchSetter(Element.prototype, 'outerHTML', 'HTML_SINK_OUTER_HTML', 'Element.outerHTML');
    if (typeof ShadowRoot !== 'undefined') {
      patchSetter(ShadowRoot.prototype, 'innerHTML', 'HTML_SINK_SHADOW_ROOT_INNER_HTML', 'ShadowRoot.innerHTML');
    }
    patchMethod(Element.prototype, 'insertAdjacentHTML', function (args) {
      add('HTML_SINK_INSERT_ADJACENT_HTML', { target: this?.tagName || '', value: args[1] });
    }, 'Element.insertAdjacentHTML');
    patchMethod(Element.prototype, 'setAttribute', function (args) {
      const attribute = String(args[0] || '').toLowerCase();
      if (attribute.startsWith('on') || ['srcdoc', 'href', 'src', 'action', 'formaction'].includes(attribute)) {
        add('ATTRIBUTE_OR_URL_SINK', { target: this?.tagName || '', attribute, value: args[1] });
      }
    }, 'Element.setAttribute');
    patchMethod(Document.prototype, 'write', function (args) {
      add('HTML_SINK_DOCUMENT_WRITE', { target: 'DOCUMENT', value: args.join('') });
    }, 'Document.write');
    patchMethod(Document.prototype, 'writeln', function (args) {
      add('HTML_SINK_DOCUMENT_WRITELN', { target: 'DOCUMENT', value: args.join('') });
    }, 'Document.writeln');
    patchMethod(globalThis.DOMParser?.prototype, 'parseFromString', function (args) {
      add('HTML_SINK_DOMPARSER', { target: 'DOMPARSER', value: args[0] });
    }, 'DOMParser.parseFromString');
    patchMethod(globalThis.Range?.prototype, 'createContextualFragment', function (args) {
      add('HTML_SINK_CONTEXTUAL_FRAGMENT', { target: 'RANGE', value: args[0] });
    }, 'Range.createContextualFragment');
    patchMethod(Element.prototype, 'setHTMLUnsafe', function (args) {
      add('HTML_SINK_SET_HTML_UNSAFE', { target: this?.tagName || '', value: args[0] });
    }, 'Element.setHTMLUnsafe');
    if (typeof ShadowRoot !== 'undefined') {
      patchMethod(ShadowRoot.prototype, 'setHTMLUnsafe', function (args) {
        add('HTML_SINK_SET_HTML_UNSAFE', { target: 'SHADOW_ROOT', value: args[0] });
      }, 'ShadowRoot.setHTMLUnsafe');
    }
    patchMethod(window, 'open', function (args) {
      const value = args[0];
      let crossOrigin = false;
      try { crossOrigin = value ? new URL(value, location.href).origin !== location.origin : false; } catch { /* invalid URL */ }
      add('WINDOW_OPEN', { target: 'WINDOW', value, crossOrigin });
    }, 'window.open');
    patchMethod(window, 'setTimeout', function (args) {
      if (typeof args[0] === 'string') add('STRING_TIMER_SINK', { target: 'WINDOW', value: args[0] });
    }, 'window.setTimeout');
    patchMethod(window, 'setInterval', function (args) {
      if (typeof args[0] === 'string') add('STRING_TIMER_SINK', { target: 'WINDOW', value: args[0] });
    }, 'window.setInterval');

    window.addEventListener('message', event => {
      const value = typeof event.data === 'string' ? event.data : JSON.stringify(Object.keys(event.data || {}));
      add('POSTMESSAGE_RECEIVED', {
        target: 'WINDOW', value, origin: event.origin,
        crossOrigin: Boolean(event.origin && event.origin !== location.origin),
      });
    }, true);

    const registerControlledMarker = event => {
      const target = event.target;
      if (!(target instanceof HTMLInputElement || target instanceof HTMLTextAreaElement || target instanceof HTMLSelectElement)) return;
      if (target instanceof HTMLInputElement && target.type === 'password') return;
      const value = String(target.value || '');
      const markers = value.match(/\bXSSSCOUT_[A-Z0-9_-]{3,128}\b/gi) || [];
      for (const marker of markers) formMarkers.add(marker);
      if (markers.length) add('FORM_CONTROLLED_MARKER', { target: target.tagName, attribute: target.name || target.id || '', value });
    };
    document.addEventListener('input', registerControlledMarker, true);
    document.addEventListener('change', registerControlledMarker, true);
    document.addEventListener('submit', event => {
      event.target?.querySelectorAll?.('input, textarea, select').forEach(control => registerControlledMarker({ target: control }));
    }, true);
    document.querySelectorAll('input, textarea, select').forEach(control => registerControlledMarker({ target: control }));

    patchMethod(window, 'fetch', function (args) {
      const [input, init = {}] = args;
      const method = init.method || (input instanceof Request ? input.method : 'GET');
      const url = input instanceof Request ? input.url : input;
      addRequestShape('FETCH', method, url, init.body);
    }, 'window.fetch');
    patchMethod(XMLHttpRequest.prototype, 'open', function (args) {
      xhrMeta.set(this, { method: args[0], url: args[1] });
    }, 'XMLHttpRequest.open');
    patchMethod(XMLHttpRequest.prototype, 'send', function (args) {
      const meta = xhrMeta.get(this) || {};
      addRequestShape('XHR', meta.method || 'GET', meta.url || location.href, args[0]);
    }, 'XMLHttpRequest.send');

    const classify = event => {
      const htmlSink = event.Kind.startsWith('HTML_SINK');
      if (htmlSink && event.SourceMatch && event.ScoutMarker) return 'HIGH';
      if (htmlSink && event.SourceMatch) return 'MEDIUM';
      if (htmlSink || event.Kind === 'STRING_TIMER_SINK') return 'LOW';
      if (event.Kind === 'WINDOW_OPEN' && event.CrossOrigin) return 'LOW';
      return 'INFO';
    };

    return {
      version: '3.3', startedAt, events, requestShapes, hookStatus,
      summary() {
        const rows = events.map(event => ({ ...event, Signal: classify(event) }));
        return {
          total: rows.length,
          high: rows.filter(row => row.Signal === 'HIGH'),
          medium: rows.filter(row => row.Signal === 'MEDIUM'),
          low: rows.filter(row => row.Signal === 'LOW'),
          info: rows.filter(row => row.Signal === 'INFO'),
          rows,
          requestShapes: [...requestShapes],
          stateChangingRequestShapes: requestShapes.filter(row => !['GET', 'HEAD', 'OPTIONS'].includes(row.Method)),
          controlledInputMarkerCount: formMarkers.size,
          hookStatus: { ...hookStatus },
          activeHookCount: Object.values(hookStatus).filter(status => status === 'ACTIVE').length,
          unavailableHooks: Object.entries(hookStatus).filter(([, status]) => status !== 'ACTIVE').map(([hook]) => hook),
        };
      },
      stop() { restores.splice(0).reverse().forEach(restore => restore()); },
    };
  })();

  // ============================================================
  // XSS SCOUT v2.8
  //
  // GENERALIZED GET REPLAY + RESPONSE STATE
  // + RESYNC CHARACTER-LEVEL TRANSFORMATION ANALYSIS

  // 핵심:
  //   1. CURRENT / NAVIGATION / RESOURCE(fetch/XHR) Discovery
  //   2. query parameter 이름 기반 제외 없음
  //   3. UNKNOWN RESOURCE → GET Baseline Probe
  //   4. parameter 하나씩 mutation
  //   5. HTTP / Content-Type / Length 변화 분석
  //   6. marker reflection 탐지
  //   7. SPECIAL 문자 단위 변환 분석

  // CONFIG
  // ============================================================

  const CONFIG = {

    SPECIAL:
      `<>;:'"\``,

    TIMEOUT:
      10000,

    LENGTH_DIFF_THRESHOLD:
      500,

    LENGTH_DIFF_RATIO_THRESHOLD:
      0.10,

    TEST_SAME_ORIGIN:
      true,

    // Active mutations never leave the current origin by default.
    // Set true only for an explicitly authorized multi-origin test scope.
    TEST_CROSS_ORIGIN:
      false,

    STATIC_EXT:
      /\.(?:js|css|png|jpe?g|gif|svg|webp|ico|woff2?|woff|ttf|eot|map|mp4|webm|mp3|pdf)$/i,

    SKIP_CONTENT_TYPE:
      /^(?:text\/css|image\/|font\/|audio\/|video\/|application\/font|application\/octet-stream)/i,

    METHOD_REJECT_STATUS:
      new Set([
        405,
        501
      ]),

      300,

    RESYNC_LOOKAHEAD:
      80
  };


  // ============================================================
  // PARAMETER
  // ============================================================

  function getParameters(url) {

    const params = [];

    for (
      const [name]
      of url.searchParams.entries()
    ) {

      if (!name) {
        continue;
      }

      params.push(name);
    }

    return [
      ...new Set(params)
    ];
  }


  function getOriginalValues(
    url,
    parameter
  ) {

    return url.searchParams.getAll(
      parameter
    );
  }


  // ============================================================
  // HTTP STATUS CLASS
  // ============================================================

  function classifyStatus(status) {

    if (
      status >= 200 &&
      status < 300
    ) {
      return 'SUCCESS';
    }

    if (
      status >= 300 &&
      status < 400
    ) {
      return 'REDIRECT';
    }

    if (
      status >= 400 &&
      status < 500
    ) {
      return 'CLIENT_ERROR';
    }

    if (
      status >= 500 &&
      status < 600
    ) {
      return 'SERVER_ERROR';
    }

    return 'OTHER';
  }


  function isErrorClass(statusClass) {

    return (
      statusClass === 'CLIENT_ERROR' ||
      statusClass === 'SERVER_ERROR'
    );
  }


  // ============================================================
  // CONTENT TYPE
  // ============================================================

  function normalizeContentType(contentType) {

    return String(
      contentType || ''
    )
      .split(';')[0]
      .trim()
      .toLowerCase();
  }


  function getContentTypeTransition(
    baselineType,
    testedType
  ) {

    const from =
      normalizeContentType(
        baselineType
      );

    const to =
      normalizeContentType(
        testedType
      );

    if (
      from === to
    ) {
      return 'STABLE';
    }

    return `${from || '(none)'} → ${to || '(none)'}`;
  }


  // ============================================================
  // FETCH
  // ============================================================

  async function fetchGET(url) {

    const controller =
      new AbortController();

    const timer =
      setTimeout(
        () => controller.abort(),
        CONFIG.TIMEOUT
      );

    try {

      return await fetch(
        url,
        {
          method:
            'GET',

          credentials:
            'include',

          cache:
            'no-store',

          redirect:
            'follow',

          signal:
            controller.signal
        }
      );

    } finally {

      clearTimeout(timer);
    }
  }


  // ============================================================
  // RESPONSE
  // ============================================================

  async function readResponse(response) {

    const contentType =
      response.headers.get(
        'content-type'
      ) || '';

    const body =
      await response.text();

    return {

      status:
        response.status,

      statusClass:
        classifyStatus(
          response.status
        ),

      ok:
        response.ok,

      redirected:
        response.redirected,

      finalURL:
        response.url,

      contentType,

      normalizedContentType:
        normalizeContentType(
          contentType
        ),

      body,

      length:
        body.length
    };
  }


  // ============================================================
  // CONTEXT
  // ============================================================

  function getContext(
    body,
    marker,
    before = 150,
    after = 300
  ) {

    const index =
      body.indexOf(marker);

    if (
      index === -1
    ) {
      return '(NOT FOUND)';
    }

    return body
      .substring(

        Math.max(
          0,
          index - before
        ),

        Math.min(
          body.length,
          index +
          marker.length +
          after
        )
      )

      .replace(/\r/g, '\\r')
      .replace(/\n/g, '\\n')
      .replace(/\t/g, '\\t');
  }


  // ============================================================
  // REFLECTION POSITIONS
  // ============================================================

  function findMarkerPositions(
    body,
    marker
  ) {

    const positions = [];

    let offset = 0;

    while (true) {

      const index =
        body.indexOf(
          marker,
          offset
        );

      if (
        index === -1
      ) {
        break;
      }

      positions.push(index);

      offset =
        index +
        marker.length;
    }

    return positions;
  }

  // ============================================================
  // REPRESENTATION LIST
  // ============================================================

  function getRepresentations(character) {

    const representations = [

      {
        status:
          'RAW',

        representation:
          character
      }
    ];

    const definitions =
      CHAR_ENCODINGS[character] || {};

    for (
      const [
        encoding,
        values
      ]
      of Object.entries(definitions)
    ) {

      for (
        const value
        of values
      ) {

        representations.push({

          status:
            encoding,

          representation:
            value
        });
      }
    }

    return representations;
  }


  // ============================================================
  // CHARACTER MATCH
  // ============================================================

  function matchCharacter(
    text,
    offset,
    character
  ) {

    const representations =
      getRepresentations(
        character
      );

    for (
      const item
      of representations
    ) {

      if (
        text.startsWith(
          item.representation,
          offset
        )
      ) {

        return {

          status:
            item.status,

          representation:
            item.representation,

          position:
            offset,

          skipped:
            0,

          consumed:
            item.representation.length
        };
      }
    }

    return null;
  }


  // ============================================================
  // RESYNC SEARCH
  // ============================================================

  function findCharacterAhead(
    text,
    offset,
    character
  ) {

    const representations =
      getRepresentations(
        character
      );

    const searchEnd =
      Math.min(
        text.length,
        offset +
        CONFIG.RESYNC_LOOKAHEAD
      );

    let best = null;

    for (
      const item
      of representations
    ) {

      const index =
        text.indexOf(
          item.representation,
          offset
        );

      if (
        index === -1 ||
        index >= searchEnd
      ) {

        continue;
      }

      if (
        !best ||
        index < best.position ||
        (
          index === best.position &&
          item.representation.length >
          best.representation.length
        )
      ) {

        best = {

          status:
            item.status,

          representation:
            item.representation,

          position:
            index,

          skipped:
            index - offset,

          consumed:
            item.representation.length
        };
      }
    }

    return best;
  }


  // ============================================================
  // NEXT CHARACTER RESYNC
  // ============================================================

  function findNextExpectedCharacter(
    text,
    offset,
    specialCharacters,
    currentIndex
  ) {

    let best = null;

    for (
      let i =
        currentIndex + 1;

      i <
        specialCharacters.length;

      i++
    ) {

      const character =
        specialCharacters[i];

      const match =
        findCharacterAhead(
          text,
          offset,
          character
        );

      if (!match) {
        continue;
      }

      if (
        !best ||
        match.position <
        best.match.position
      ) {

        best = {

          characterIndex:
            i,

          character,

          match
        };
      }
    }

    return best;
  }


  // ============================================================
  // RESYNC CHARACTER ANALYSIS
  // ============================================================

  function analyzeReflectionAt(
    body,
    marker,
    markerPosition
  ) {

    const start =
      markerPosition +
      marker.length;

    const segment =
      body.substring(
        start,
        Math.min(
          body.length,
          start +
          CONFIG.TRANSFORM_LOOKAHEAD
        )
      );

    const characters =
      [...CONFIG.SPECIAL];

    const map = [];

    let offset = 0;

    let resyncCount = 0;


    for (
      let i = 0;
      i < characters.length;
      i++
    ) {

      const character =
        characters[i];


      // ----------------------------------------------------------
      // 1. 현재 위치에서 정상 match
      // ----------------------------------------------------------

      const direct =
        matchCharacter(
          segment,
          offset,
          character
        );


      if (
        direct
      ) {

        map.push({

          Character:
            character,

          Status:
            direct.status,

          Representation:
            direct.representation,

          Position:
            direct.position,

          Skipped:
            0,

          Resynced:
            false
        });


        offset =
          direct.position +
          direct.consumed;


        continue;
      }


      // ----------------------------------------------------------
      // 2. 현재 문자가 약간 뒤에서 발견되는지 확인
      //
      // 단순히 뒤에서 발견됐다고 바로 해당 문자라고 판단하면
      // 주변 HTML/JSON 문자와 혼동할 수 있으므로,
      // RESYNC_LOOKAHEAD 안에서만 제한적으로 탐색한다.
      // ----------------------------------------------------------

      const ahead =
        findCharacterAhead(
          segment,
          offset,
          character
        );


      // ----------------------------------------------------------
      // 3. 다음 기대 문자 위치도 확인
      // ----------------------------------------------------------

      const nextExpected =
        findNextExpectedCharacter(
          segment,
          offset,
          characters,
          i
        );


      /*
       * 현재 문자의 representation이 먼저 나온다면:
       *
       *   unexpected bytes + current character
       *
       * 로 판단하고 현재 문자에 resync.
       */

      if (
        ahead &&
        (
          !nextExpected ||
          ahead.position <
          nextExpected.match.position
        )
      ) {

        map.push({

          Character:
            character,

          Status:
            ahead.status,

          Representation:
            ahead.representation,

          Position:
            ahead.position,

          Skipped:
            ahead.skipped,

          Resynced:
            true
        });


        offset =
          ahead.position +
          ahead.consumed;


        resyncCount++;


        continue;
      }


      /*
       * 다음 기대 문자가 먼저 발견되면
       * 현재 문자는 응답 sequence에서 사라진 것으로 본다.
       *
       * REMOVED라고 확정하지 않고 MISSING으로만 기록한다.
       */

      if (
        nextExpected
      ) {

        map.push({

          Character:
            character,

          Status:
            'MISSING',

          Representation:
            '',

          Position:
            offset,

          Skipped:
            nextExpected.match.position -
            offset,

          Resynced:
            true
        });


        /*
         * offset은 건드리지 않는다.
         *
         * 다음 loop에서 next character가 현재 offset 이후에서
         * 다시 정상적으로 탐색된다.
         */

        resyncCount++;


        continue;
      }


      // ----------------------------------------------------------
      // 4. 이후 기대 문자도 발견되지 않음
      // ----------------------------------------------------------

      map.push({

        Character:
          character,

        Status:
          'MISSING_OR_TRUNCATED',

        Representation:
          segment.substring(
            offset,
            offset + 20
          ),

        Position:
          offset,

        Skipped:
          0,

        Resynced:
          false
      });
    }


    return {

      markerPosition,

      reflectedSegment:
        segment.substring(
          0,
          Math.min(
            segment.length,
            150
          )
        ),

      resyncCount,

      charMap:
        map
    };
  }


  // ============================================================
  // AGGREGATE CHARACTER MAP
  // ============================================================

  function aggregateCharMap(
    reflectionAnalyses
  ) {

    const aggregate = {};


    for (
      const character
      of CONFIG.SPECIAL
    ) {

      aggregate[character] = {

        RAW:
          0,

        HTML_ENTITY:
          0,

        JSON_ESCAPE:
          0,

        JS_ESCAPE:
          0,

        URL_ENCODED:
          0,

        MISSING:
          0,

        MISSING_OR_TRUNCATED:
          0
      };
    }


    for (
      const analysis
      of reflectionAnalyses
    ) {

      for (
        const item
        of analysis.charMap
      ) {

        if (
          aggregate[item.Character] &&
          aggregate[item.Character][item.Status] !== undefined
        ) {

          aggregate[item.Character][item.Status]++;
        }
      }
    }


    return aggregate;
  }


  // ============================================================
  // TRANSFORMATION CLASS
  // ============================================================

  function classifyTransformation(
    reflectionAnalyses
  ) {

    if (
      !reflectionAnalyses.length
    ) {

      return 'NOT_REFLECTED';
    }


    const statuses =
      reflectionAnalyses.flatMap(
        analysis =>
          analysis.charMap.map(
            item =>
              item.Status
          )
      );


    if (
      statuses.length &&
      statuses.every(
        status =>
          status === 'RAW'
      )
    ) {

      return 'RAW';
    }


    const rawCount =
      statuses.filter(
        status =>
          status === 'RAW'
      ).length;


    const encodedStatuses =
      statuses.filter(
        status =>
          status === 'HTML_ENTITY' ||
          status === 'JSON_ESCAPE' ||
          status === 'JS_ESCAPE' ||
          status === 'URL_ENCODED'
      );


    const missingStatuses =
      statuses.filter(
        status =>
          status === 'MISSING' ||
          status === 'MISSING_OR_TRUNCATED'
      );


    if (
      encodedStatuses.length &&
      !rawCount &&
      !missingStatuses.length
    ) {

      const encodingTypes =
        new Set(
          encodedStatuses
        );


      if (
        encodingTypes.size === 1
      ) {

        return [
          ...encodingTypes
        ][0];
      }


      return 'FULL_ENCODED_MIXED';
    }


    if (
      encodedStatuses.length &&
      rawCount &&
      !missingStatuses.length
    ) {

      return 'PARTIAL_ENCODED';
    }


    if (
      missingStatuses.length &&
      !rawCount &&
      !encodedStatuses.length
    ) {

      return 'FILTERED_OR_TRUNCATED';
    }


    if (
      missingStatuses.length &&
      (
        rawCount ||
        encodedStatuses.length
      )
    ) {

      return 'PARTIAL_FILTERED';
    }


    return 'OTHER_TRANSFORM';
  }


  // ============================================================
  // CHARACTER SUMMARY
  // ============================================================

  function getCharacterSummary(
    reflectionAnalyses
  ) {

    if (
      !reflectionAnalyses.length
    ) {

      return '-';
    }


    const first =
      reflectionAnalyses[0];


    return first.charMap
      .map(
        item => {

          const suffix =
            item.Resynced
              ? `@+${item.Skipped}`
              : '';

          return (
            `${item.Character}:${item.Status}${suffix}`
          );
        }
      )
      .join(' | ');
  }


  // ============================================================
  // HTML CONTEXT
  // ============================================================

  function classifyHtmlContext(
    body,
    marker
  ) {

    const index =
      body.indexOf(marker);


    if (
      index === -1
    ) {

      return 'NOT_REFLECTED';
    }


    const before =
      body.substring(
        0,
        index
      );


    const lower =
      before.toLowerCase();


    const scriptStart =
      lower.lastIndexOf('<script');

    if (
      scriptStart >
      lower.lastIndexOf('</script')
    ) {

      const tagEnd =
        lower.indexOf('>', scriptStart);

      const openingTag =
        tagEnd === -1
          ? lower.substring(scriptStart)
          : lower.substring(scriptStart, tagEnd + 1);

      if (
        /type\s*=\s*["']?(?:application\/(?:ld\+)?json|importmap|speculationrules)/i.test(
          openingTag
        )
      ) {

        return 'SCRIPT_DATA';
      }

      return 'SCRIPT';
    }


    if (
      lower.lastIndexOf('<style') >
      lower.lastIndexOf('</style')
    ) {

      return 'STYLE';
    }


    const lastLT =
      before.lastIndexOf('<');


    const lastGT =
      before.lastIndexOf('>');


    if (
      lastLT >
      lastGT
    ) {

      const fragment =
        before.substring(
          lastLT
        );


      const dq =
        (
          fragment.match(/"/g) ||
          []
        ).length;


      const sq =
        (
          fragment.match(/'/g) ||
          []
        ).length;


      if (
        dq % 2 === 1
      ) {

        return 'HTML_ATTRIBUTE_DQ';
      }


      if (
        sq % 2 === 1
      ) {

        return 'HTML_ATTRIBUTE_SQ';
      }


      return 'HTML_ATTRIBUTE_UNQUOTED';
    }


    return 'HTML_TEXT';
  }


  // ============================================================
  // CONTEXT CLASSIFIER
  // ============================================================

  function classifyContext(
    body,
    marker,
    contentType
  ) {

    if (
      !body.includes(marker)
    ) {

      return 'NOT_REFLECTED';
    }


    const type =
      normalizeContentType(
        contentType
      );


    if (
      type.includes('json') ||
      type.includes('+json')
    ) {

      return 'JSON';
    }


    if (
      type.includes('javascript') ||
      type.includes('ecmascript')
    ) {

      return 'SCRIPT';
    }


    if (
      type.includes('html') ||
      type.includes('xhtml')
    ) {

      return classifyHtmlContext(
        body,
        marker
      );
    }


    if (
      type.startsWith('text/')
    ) {

      return 'TEXT';
    }


    return 'UNKNOWN';
  }


  // ============================================================
  // VALUE TYPE
  // ============================================================

  function inferValueType(value) {

    if (
      value === ''
    ) {

      return 'EMPTY';
    }


    if (
      /^(?:true|false)$/i.test(value)
    ) {

      return 'BOOLEAN';
    }


    if (
      /^-?(?:\d+|\d*\.\d+)$/.test(value)
    ) {

      return 'NUMBER';
    }


    return 'STRING';
  }


  // ============================================================
  // LENGTH ANALYSIS
  // ============================================================

  function getLengthAnalysis(
    baselineLength,
    testedLength
  ) {

    const diff =
      testedLength -
      baselineLength;


    const absoluteDiff =
      Math.abs(
        diff
      );


    const ratio =
      baselineLength > 0
        ? absoluteDiff /
          baselineLength
        : (
            absoluteDiff > 0
              ? 1
              : 0
          );


    const significant =
      absoluteDiff >
        CONFIG.LENGTH_DIFF_THRESHOLD &&

      ratio >
        CONFIG.LENGTH_DIFF_RATIO_THRESHOLD;


    return {

      diff,

      ratio,

      significant
    };
  }


  // ============================================================
  // RESULT CLASSIFIER
  // ============================================================

  function classifyResult({

    baseline,

    tested,

    reflected,

    exact,

    lengthAnalysis

  }) {

    const statusChanged =
      baseline.status !==
      tested.status;


    const classChanged =
      baseline.statusClass !==
      tested.statusClass;


    const baselineError =
      isErrorClass(
        baseline.statusClass
      );


    const testedError =
      isErrorClass(
        tested.statusClass
      );


    if (
      reflected
    ) {

      if (
        !baselineError &&
        testedError
      ) {

        return 'ERROR_REFLECTION';
      }


      if (
        baselineError
      ) {

        return 'ERROR_BASELINE_REFLECTION';
      }


      if (
        statusChanged ||
        classChanged
      ) {

        return 'STATUS_CHANGED_REFLECTION';
      }


      if (
        exact
      ) {

        return 'STABLE_EXACT_REFLECTION';
      }


      return 'STABLE_TRANSFORMED_REFLECTION';
    }


    if (
      statusChanged ||
      classChanged
    ) {

      return 'VALIDATION_STATUS_CHANGE';
    }


    if (
      lengthAnalysis.significant
    ) {

      return 'RESPONSE_CHANGED';
    }


    return 'NO_REFLECTION';
  }

  // Resource Timing may be unavailable or cleared by the application. Query
  // URLs already present in same-page link/form markup are safe discovery
  // candidates: this step observes them only; GET replay remains gated by the
  // existing same-origin policy below.
  //
  // GET <form> fields are a common reflected-XSS vector (search boxes, filter
  // forms) but a form's own `.action` property NEVER includes its input
  // values -- those only become a query string once the browser actually
  // submits the form. Without synthesizing that query string here, every GET
  // form's parameters were silently invisible to discovery (url.search was
  // always empty, so the whole form got dropped by the `!url.search` filter
  // in the DISCOVERY step below). Also switched the selector from
  // `form[action]` to plain `form`, since a form with no `action` attribute
  // at all (submits to the current page, a very common pattern) has no
  // `action` attribute to match on but is still a perfectly valid GET form.
  const buildFormGetURL = form => {
    const actionAttr = form.getAttribute('action');
    const url = new URL(actionAttr || location.href, location.href);
    const params = new URLSearchParams(url.search);
    form.querySelectorAll('input[name], select[name], textarea[name]').forEach(field => {
      if (field.disabled) return;
      const type = (field.type || '').toLowerCase();
      if (type === 'submit' || type === 'button' || type === 'reset' || type === 'file' || type === 'image') return;
      if ((type === 'checkbox' || type === 'radio') && !field.checked) return;
      params.set(field.name, field.value ?? '');
    });
    url.search = params.toString();
    return url.href;
  };

  // XSS SCOUT v2.9
  // PASSIVE RESULT ANALYZER
  //
  // 입력:
  //   window.__XSS_SCOUT__.results
  //
  // 기능:
  //   - 추가 HTTP 요청 없음
  //   - HIGH / MEDIUM / LOW / INFO signal 분류
  //   - endpoint 단위 saturation 계산
  //   - 반복적인 FULL URL encoding noise 압축
  //   - 원본 results 보존
  // ============================================================


  const scout =
    window.__XSS_SCOUT__;


  if (
    !scout ||
    !Array.isArray(scout.results)
  ) {

    console.error(
      '[XSS Scout v2.9] v2.8 results가 없습니다.'
    );

    return;
  }


  const results =
    scout.results;


  // ============================================================
  // HELPERS
  // ============================================================

  function endpointKey(result) {

    return [
      result.Origin || '',
      result.Host || '',
      result.Path || ''
    ].join('|');
  }


  function isErrorReflection(result) {

    return (
      result.Result === 'ERROR_REFLECTION' ||
      result.Result === 'ERROR_BASELINE_REFLECTION'
    );
  }


  function isStableReflection(result) {

    return (
      result.Result === 'STABLE_EXACT_REFLECTION' ||
      result.Result === 'STABLE_TRANSFORMED_REFLECTION'
    );
  }


  function isFilteredTransform(result) {

    return (
      result.Transformation === 'PARTIAL_FILTERED' ||
      result.Transformation === 'FILTERED_OR_TRUNCATED'
    );
  }


  function isFullUrlEncoded(result) {

    const map =
      result.CharacterMap;


    if (!map) {
      return false;
    }


    const special =
      [...`<>;:'"\``];


    let seen = 0;


    for (
      const character
      of special
    ) {

      const entry =
        map[character];


      if (!entry) {
        return false;
      }


      const total =
        Object.values(entry)
          .reduce(
            (sum, value) =>
              sum +
              (
                Number(value) || 0
              ),
            0
          );


      if (
        total === 0
      ) {

        return false;
      }


      if (
        (
          entry.URL_ENCODED || 0
        ) !== total
      ) {

        return false;
      }


      seen++;
    }


    return (
      seen ===
      special.length
    );
  }


  function hasRawCharacter(result) {

    const map =
      result.CharacterMap;


    if (!map) {
      return false;
    }


    return Object
      .values(map)
      .some(
        entry =>
          (
            entry?.RAW || 0
          ) > 0
      );
  }


  function hasRawStructuralCharacter(result) {

    const map =
      result.CharacterMap;


    if (!map) {
      return false;
    }


    return [
      '<',
      '>',
      '"',
      "'"
    ].some(
      character =>
        (
          map[character]?.RAW ||
          0
        ) > 0
    );
  }


  function hasMissingCharacter(result) {

    const map =
      result.CharacterMap;


    if (!map) {
      return false;
    }


    return Object
      .values(map)
      .some(
        entry =>
          (
            (
              entry?.MISSING || 0
            ) +

            (
              entry?.MISSING_OR_TRUNCATED || 0
            )
          ) > 0
      );
  }


  // ============================================================
  // ENDPOINT GROUPING
  // ============================================================

  const endpointMap =
    new Map();


  for (
    const result
    of results
  ) {

    const key =
      endpointKey(
        result
      );


    if (
      !endpointMap.has(key)
    ) {

      endpointMap.set(
        key,
        {
          key,

          Origin:
            result.Origin,

          Host:
            result.Host,

          Path:
            result.Path,

          results:
            []
        }
      );
    }


    endpointMap
      .get(key)
      .results
      .push(result);
  }


  // ============================================================
  // ENDPOINT STATISTICS
  // ============================================================

  const endpointStats = [];


  for (
    const endpoint
    of endpointMap.values()
  ) {

    const rows =
      endpoint.results;


    const tested =
      rows.length;


    const reflected =
      rows.filter(
        row =>
          row.Reflected
      ).length;


    const stable =
      rows.filter(
        isStableReflection
      ).length;


    const errors =
      rows.filter(
        isErrorReflection
      ).length;


    const raw =
      rows.filter(
        row =>
          row.Transformation === 'RAW'
      ).length;


    const partial =
      rows.filter(
        row =>
          row.Transformation === 'PARTIAL_ENCODED'
      ).length;


    const filtered =
      rows.filter(
        isFilteredTransform
      ).length;


    const fullUrlEncoded =
      rows.filter(
        row =>
          row.Reflected &&
          isFullUrlEncoded(row)
      ).length;


    const reflectionRate =
      tested
        ? reflected / tested
        : 0;


    const fullUrlEncodingRate =
      reflected
        ? fullUrlEncoded / reflected
        : 0;


    // ----------------------------------------------------------
    // DOMINANT TRANSFORMATION
    // ----------------------------------------------------------

    const transformCounts =
      new Map();


    for (
      const row
      of rows
    ) {

      if (
        !row.Reflected
      ) {
        continue;
      }


      const type =
        row.Transformation ||
        'UNKNOWN';


      transformCounts.set(
        type,
        (
          transformCounts.get(type) ||
          0
        ) + 1
      );
    }


    const dominantTransform =
      [
        ...transformCounts.entries()
      ]

        .sort(
          (a, b) =>
            b[1] - a[1]
        )[0]?.[0] ||
        '-';


    // ----------------------------------------------------------
    // SATURATION
    // ----------------------------------------------------------

    let saturation =
      'NONE';


    if (
      tested >= 3 &&
      reflectionRate >= 0.90
    ) {

      saturation =
        'VERY_HIGH';

    } else if (
      tested >= 3 &&
      reflectionRate >= 0.60
    ) {

      saturation =
        'HIGH';

    } else if (
      reflectionRate > 0
    ) {

      saturation =
        'PARTIAL';
    }


    // ----------------------------------------------------------
    // HOMOGENEOUS URL ENCODING
    // ----------------------------------------------------------

    const homogeneousUrlEncoding =
      (
        reflected >= 3 &&
        fullUrlEncodingRate >= 0.90
      );


    endpointStats.push({

      Key:
        endpoint.key,

      Origin:
        endpoint.Origin,

      Host:
        endpoint.Host,

      Path:
        endpoint.Path,

      Tested:
        tested,

      Reflected:
        reflected,

      ReflectionRate:
        Number(
          reflectionRate
            .toFixed(3)
        ),

      Stable:
        stable,

      ErrorReflection:
        errors,

      Raw:
        raw,

      PartialEncoded:
        partial,

      Filtered:
        filtered,

      FullUrlEncoded:
        fullUrlEncoded,

      FullUrlEncodingRate:
        Number(
          fullUrlEncodingRate
            .toFixed(3)
        ),

      DominantTransform:
        dominantTransform,

      Saturation:
        saturation,

      HomogeneousUrlEncoding:
        homogeneousUrlEncoding
    });
  }


  // ============================================================
  // LOOKUP
  // ============================================================

  const endpointLookup =
    new Map(
      endpointStats.map(
        stat => [
          stat.Key,
          stat
        ]
      )
    );


  // ============================================================
  // SIGNAL CLASSIFICATION
  // ============================================================

  function classifySignal(result) {

    const endpoint =
      endpointLookup.get(
        endpointKey(result)
      );


    if (
      !result.Reflected
    ) {

      if (
        result.Result ===
        'VALIDATION_STATUS_CHANGE'
      ) {

        return {
          Signal:
            'INFO',

          Reason:
            'VALIDATION_CHANGE'
        };
      }


      if (
        result.Result ===
        'RESPONSE_CHANGED'
      ) {

        return {
          Signal:
            'INFO',

          Reason:
            'RESPONSE_CHANGE'
        };
      }


      return {
        Signal:
          'INFO',

        Reason:
          'NO_REFLECTION'
      };
    }


    // ----------------------------------------------------------
    // ERROR REFLECTION
    // ----------------------------------------------------------

    if (
      isErrorReflection(
        result
      )
    ) {

      return {
        Signal:
          'LOW',

        Reason:
          'ERROR_RESPONSE_REFLECTION'
      };
    }


    // ----------------------------------------------------------
    // NON-EXECUTABLE SCRIPT DATA
    //
    // application/json, import maps, and speculation rules are serialized
    // data containers, not executable JavaScript. Preserve the reflection as
    // a reviewable fact without promoting it to an XSS finding.
    // ----------------------------------------------------------

    if (
      result.ContextType ===
      'SCRIPT_DATA'
    ) {

      return {
        Signal:
          'LOW',

        Reason:
          'SERIALIZED_SCRIPT_DATA_REFLECTION'
      };
    }


    // ----------------------------------------------------------
    // SERIALIZED JSON
    //
    // A response declared as JSON is data, not an HTML execution context.
    // Keep it visible for data-flow review, but do not promote reflection in
    // the response itself to a direct XSS finding.
    // ----------------------------------------------------------

    if (
      result.ContextType ===
      'JSON'
    ) {

      return {
        Signal:
          'LOW',

        Reason:
          'SERIALIZED_JSON_REFLECTION'
      };
    }


    // ----------------------------------------------------------
    // SCRIPT CONTEXT WITH TRANSFORMATION
    //
    // "SCRIPT" also covers serialized bootstrap/config data. A transformed
    // reflection alone is insufficient to claim executable JavaScript, so it
    // is a manual-review candidate rather than an automatic HIGH.
    // ----------------------------------------------------------

    if (
      result.ContextType === 'SCRIPT' &&
      (
        result.Transformation === 'PARTIAL_FILTERED' ||
        result.Transformation === 'PARTIAL_ENCODED'
      )
    ) {

      return {
        Signal:
          'MEDIUM',

        Reason:
          'SCRIPT_TRANSFORM_REVIEW'
      };
    }


    // ----------------------------------------------------------
    // FILTERING / CHARACTER REMOVAL
    // ----------------------------------------------------------

    if (
      isFilteredTransform(
        result
      ) ||
      hasMissingCharacter(
        result
      )
    ) {

      return {
        Signal:
          'HIGH',

        Reason:
          'CHARACTER_FILTERING'
      };
    }


    // ----------------------------------------------------------
    // RAW
    // ----------------------------------------------------------

    if (
      result.Transformation ===
      'RAW'
    ) {

      return {
        Signal:
          'HIGH',

        Reason:
          'RAW_REFLECTION'
      };
    }


    // ----------------------------------------------------------
    // PARTIAL ENCODING + RAW CHARACTER SURVIVAL
    // ----------------------------------------------------------

    if (
      result.Transformation ===
        'PARTIAL_ENCODED' &&
      result.ContextType ===
        'HTML_TEXT' &&
      hasRawCharacter(result) &&
      !hasRawStructuralCharacter(result)
    ) {

      return {
        Signal:
          'LOW',

        Reason:
          'HTML_TEXT_STRUCTURAL_ENCODING'
      };
    }

    if (
      result.Transformation ===
        'PARTIAL_ENCODED' &&
      hasRawCharacter(result)
    ) {

      return {
        Signal:
          'HIGH',

        Reason:
          'PARTIAL_ENCODING'
      };
    }


    // ----------------------------------------------------------
    // SATURATED FULL URL ENCODING
    // ----------------------------------------------------------

    if (
      endpoint &&
      endpoint.HomogeneousUrlEncoding &&
      isFullUrlEncoded(result)
    ) {

      return {
        Signal:
          'LOW',

        Reason:
          'SATURATED_FULL_URL_ENCODING'
      };
    }


    // ----------------------------------------------------------
    // FULL URL ENCODING
    // ----------------------------------------------------------

    if (
      isFullUrlEncoded(
        result
      )
    ) {

      return {
        Signal:
          'LOW',

        Reason:
          'FULL_URL_ENCODING'
      };
    }


    // ----------------------------------------------------------
    // CONTEXT-SENSITIVE TRANSFORMATION
    // ----------------------------------------------------------

    if (
      result.Transformation ===
        'HTML_ENTITY' ||

      result.Transformation ===
        'JSON_ESCAPE' ||

      result.Transformation ===
        'FULL_ENCODED_MIXED'
    ) {

      return {
        Signal:
          'MEDIUM',

        Reason:
          'CONTEXT_ENCODING'
      };
    }


    // ----------------------------------------------------------
    // DEFAULT REFLECTION
    // ----------------------------------------------------------

    return {
      Signal:
        'MEDIUM',

      Reason:
        'REFLECTION_REVIEW'
    };
  }


  // ============================================================
  // ENRICH RESULTS
  // ============================================================

  const analyzedResults =
    results.map(
      result => {

        const signal =
          classifySignal(
            result
          );


        const endpoint =
          endpointLookup.get(
            endpointKey(result)
          );


        return {

          ...result,

          Signal:
            signal.Signal,

          SignalReason:
            signal.Reason,

          EndpointReflectionRate:
            endpoint
              ?.ReflectionRate ??
            0,

          EndpointSaturation:
            endpoint
              ?.Saturation ??
            'NONE',

          EndpointDominantTransform:
            endpoint
              ?.DominantTransform ??
            '-'
        };
      }
    );


  // ============================================================
  // PRIORITY ORDER
  // ============================================================

  const signalWeight = {

    HIGH:
      4,

    MEDIUM:
      3,

    LOW:
      2,

    INFO:
      1
  };


  const priorityResults =
    [...analyzedResults]

      .sort(
        (a, b) => {

          const signalDiff =
            (
              signalWeight[b.Signal] ||
              0
            ) -
            (
              signalWeight[a.Signal] ||
              0
            );


          if (
            signalDiff !== 0
          ) {

            return signalDiff;
          }


          if (
            a.Reflected !==
            b.Reflected
          ) {

            return Number(
              b.Reflected
            ) -
            Number(
              a.Reflected
            );
          }


          return (
            (
              b.EndpointReflectionRate ||
              0
            ) -
            (
              a.EndpointReflectionRate ||
              0
            )
          );
        }
      );


  // ============================================================
  // GROUPS
  // ============================================================

  const highSignal =
    priorityResults.filter(
      row =>
        row.Signal === 'HIGH'
    );


  const mediumSignal =
    priorityResults.filter(
      row =>
        row.Signal === 'MEDIUM'
    );


  const lowSignal =
    priorityResults.filter(
      row =>
        row.Signal === 'LOW'
    );


  const infoSignal =
    priorityResults.filter(
      row =>
        row.Signal === 'INFO'
    );


  // ============================================================
  // ENDPOINT SUMMARY
  // ============================================================

  console.log('');

  console.log(
    '================ XSS Scout -- 결과 분석(위험도 분류) ================'
  );


  console.log('');

  console.log(
    '================ 엔드포인트별 요약 ================'
  );


  console.table(

    [...endpointStats]

      .sort(
        (a, b) => {

          if (
            a.HomogeneousUrlEncoding !==
            b.HomogeneousUrlEncoding
          ) {

            return Number(
              a.HomogeneousUrlEncoding
            ) -
            Number(
              b.HomogeneousUrlEncoding
            );
          }


          return (
            b.ReflectionRate -
            a.ReflectionRate
          );
        }
      )

      .map(
        endpoint => ({

          출처:
            endpoint.Origin,

          호스트:
            endpoint.Host,

          경로:
            endpoint.Path,

          테스트횟수:
            endpoint.Tested,

          반사횟수:
            endpoint.Reflected,

          반사율:
            endpoint.ReflectionRate,

          주요변환유형:
            endpoint.DominantTransform,

          포화도:
            endpoint.Saturation,

          전체URL인코딩횟수:
            endpoint.FullUrlEncoded,

          URL인코딩비율:
            endpoint.FullUrlEncodingRate,

          반복적URL인코딩:
            endpoint.HomogeneousUrlEncoding
        })
      )
  );


  // ============================================================
  // PRIORITY RESULTS
  // ============================================================

  console.log('');

  console.log(
    '================ 우선순위 결과 (위험도 높은 순) ================'
  );


  console.table(

    priorityResults.map(
      row => ({

        위험도:
          row.Signal,

        판정근거:
          row.SignalReason,

        출처:
          row.Origin,

        경로:
          row.Path,

        파라미터:
          row.Parameter,

        주입한페이로드:
          row.Payload,

        '발견된조각(응답내용)':
          row.Context,

        'HTTP(기준→테스트)':
          row.HTTP,

        반사여부:
          row.Reflected,

        인코딩변환:
          row.Transformation,

        반사위치유형:
          row.ContextType,

        엔드포인트포화도:
          row.EndpointSaturation,

        엔드포인트반사율:
          row.EndpointReflectionRate,

        판정:
          row.Result,

        전체요청URL:
          row.URL
      })
    )
  );


  // ============================================================
  // HIGH SIGNAL
  // ============================================================

  console.log('');

  console.log(
    '================ 위험도 HIGH -- 가장 먼저 수동 검증할 것 ================'
  );


  if (
    highSignal.length
  ) {

    console.table(

      highSignal.map(
        row => ({

          경로:
            row.Path,

          파라미터:
            row.Parameter,

          주입한페이로드:
            row.Payload,

          '발견된조각(응답내용)':
            row.Context,

          'HTTP(기준→테스트)':
            row.HTTP,

          판정근거:
            row.SignalReason,

          인코딩변환:
            row.Transformation,

          반사위치유형:
            row.ContextType,

          문자별변환요약:
            row.CharacterSummary,

          판정:
            row.Result,

          전체요청URL:
            row.URL
        })
      )
    );

  } else {

    console.log(
      '[XSS Scout] HIGH 위험도 없음'
    );
  }


  // ============================================================
  // MEDIUM SIGNAL
  // ============================================================

  if (
    mediumSignal.length
  ) {

    console.log('');

    console.log(
      '================ 위험도 MEDIUM -- 다음으로 검증할 것 ================'
    );


    console.table(

      mediumSignal.map(
        row => ({

          경로:
            row.Path,

          파라미터:
            row.Parameter,

          주입한페이로드:
            row.Payload,

          '발견된조각(응답내용)':
            row.Context,

          'HTTP(기준→테스트)':
            row.HTTP,

          판정근거:
            row.SignalReason,

          인코딩변환:
            row.Transformation,

          반사위치유형:
            row.ContextType,

          판정:
            row.Result,

          전체요청URL:
            row.URL
        })
      )
    );
  }


  // ============================================================
  // LOW SIGNAL SUMMARY
  // ============================================================

  console.log('');

  console.log(
    '================ 위험도 LOW 요약 (엔드포인트별 묶음) ================'
  );


  const lowEndpointMap =
    new Map();


  for (
    const row
    of lowSignal
  ) {

    const key =
      endpointKey(
        row
      );


    if (
      !lowEndpointMap.has(key)
    ) {

      lowEndpointMap.set(
        key,
        {
          Origin:
            row.Origin,

          Host:
            row.Host,

          Path:
            row.Path,

          Count:
            0,

          Parameters:
            [],

          Reasons:
            new Set(),

          Transforms:
            new Set()
        }
      );
    }


    const group =
      lowEndpointMap.get(key);


    group.Count++;

    group.Parameters.push(
      row.Parameter
    );

    group.Reasons.add(
      row.SignalReason
    );

    group.Transforms.add(
      row.Transformation
    );
  }


  console.table(

    [
      ...lowEndpointMap.values()
    ].map(
      group => ({

        출처:
          group.Origin,

        호스트:
          group.Host,

        경로:
          group.Path,

        건수:
          group.Count,

        파라미터목록:
          group.Parameters.join(', '),

        판정근거목록:
          [
            ...group.Reasons
          ].join(', '),

        변환유형목록:
          [
            ...group.Transforms
          ].join(', ')
      })
    )
  );


  // ============================================================
  // SUMMARY
  // ============================================================

  console.log('');

  console.log(
    '================ 최종 요약 ================'
  );


  console.table([{

    엔드포인트수:
      endpointStats.length,

    전체결과수:
      analyzedResults.length,

    반사발견수:
      analyzedResults.filter(
        row =>
          row.Reflected
      ).length,

    HIGH:
      highSignal.length,

    MEDIUM:
      mediumSignal.length,

    LOW:
      lowSignal.length,

    INFO:
      infoSignal.length,

    포화된엔드포인트수:
      endpointStats.filter(
        endpoint =>
          endpoint.Saturation ===
            'VERY_HIGH' ||

          endpoint.Saturation ===
            'HIGH'
      ).length,

    반복URL인코딩엔드포인트수:
      endpointStats.filter(
        endpoint =>
          endpoint.HomogeneousUrlEncoding
      ).length
  }]);


  // ============================================================
  // SAVE
  // ============================================================

  window.__XSS_SCOUT_V29__ = {

    version:
      '2.9',

    sourceVersion:
      scout.version,

    endpointStats,

    results:
      analyzedResults,

    priorityResults,

    highSignal,

    mediumSignal,

    lowSignal,

    infoSignal
  };


  console.log('');

  console.log(
    '[XSS Scout] 2단계(위험도 분석) 완료'
  );


  console.log(
    'HIGH 목록:',
    'window.__XSS_SCOUT_V29__.highSignal'
  );


  console.log(
    'MEDIUM 목록:',
    'window.__XSS_SCOUT_V29__.mediumSignal'
  );


  console.log(
    'LOW 목록:',
    'window.__XSS_SCOUT_V29__.lowSignal'
  );


  console.log(
    '엔드포인트별 통계:',
    'window.__XSS_SCOUT_V29__.endpointStats'
  );


  console.log(
    '전체(우선순위순):',
    'window.__XSS_SCOUT_V29__.priorityResults'
  );

})();
